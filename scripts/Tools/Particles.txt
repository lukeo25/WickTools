<script>
(function(){
  'use strict';

  var managed = null;
  var sourceJSON = [];
  var sourceBaseTransforms = [];
  var internalLayer = null;
  var busy = false;
  var debounceTimer = 0;

  var ids = ['count','duration','lifetime','direction','spread','emitterWidth','emitterHeight','seed','speed','speedVar','gravityX','gravityY','drag','scaleStart','scaleEnd','scaleVar','opacityStart','opacityEnd','rotationStart','spin','randomRotation','stagger','loopBirths'];
  var el = {};
  for(var i=0;i<ids.length;i++) el[ids[i]] = document.getElementById(ids[i]);
  el.status = document.getElementById('status');
  el.bindBtn = document.getElementById('bindBtn');
  el.refreshBtn = document.getElementById('refreshBtn');
  el.rebindBtn = document.getElementById('rebindBtn');
  el.randomSeedBtn = document.getElementById('randomSeedBtn');

  function setStatus(msg, bad){
    el.status.textContent = msg;
    el.status.style.borderColor = bad ? '#8a4d57' : '#2d435d';
    el.status.style.color = bad ? '#ffd7de' : '#cfe3f5';
  }

  function getEditor(){
    try{ if(window.parent && window.parent.wickEditor) return window.parent.wickEditor; }catch(e1){}
    try{ if(window.wickEditor) return window.wickEditor; }catch(e2){}
    return null;
  }

  function getProject(){
    var ed = getEditor();
    try{ if(ed && ed.project) return ed.project; }catch(e1){}
    try{ if(window.parent && window.parent.project) return window.parent.project; }catch(e2){}
    return null;
  }

  function getWick(){
    try{ if(window.parent && window.parent.Wick) return window.parent.Wick; }catch(e1){}
    try{ if(window.Wick) return window.Wick; }catch(e2){}
    return null;
  }

  function selectedObjects(project){
    try{
      if(project && project.selection && typeof project.selection.getSelectedObjects === 'function'){
        return (project.selection.getSelectedObjects() || []).slice();
      }
    }catch(e){}
    return [];
  }

  function isPath(o){
    if(!o) return false;
    var c = String(o.classname || o.className || (o.constructor && o.constructor.name) || '').toLowerCase();
    if(c.indexOf('path') >= 0) return true;
    return !!(o.view && o.view.item && !o.timeline);
  }

  function clonePlain(v){
    try{return JSON.parse(JSON.stringify(v));}catch(e){return v;}
  }

  function exportObjectJSON(obj){
    if(!obj) return null;
    try{ if(typeof obj.exportJSON === 'function') return clonePlain(obj.exportJSON({asString:false})); }catch(e1){}
    try{ if(obj.json) return clonePlain(obj.json); }catch(e2){}
    return null;
  }

  function readTransform(obj){
    var t = {};
    try{ if(obj && obj.transformation) t = obj.transformation; }catch(e0){}
    function n(v,d){v=Number(v);return isFinite(v)?v:d;}
    return {
      x:n(t.x != null ? t.x : obj.x,0),
      y:n(t.y != null ? t.y : obj.y,0),
      scaleX:n(t.scaleX != null ? t.scaleX : obj.scaleX,1),
      scaleY:n(t.scaleY != null ? t.scaleY : obj.scaleY,1),
      rotation:n(t.rotation != null ? t.rotation : obj.rotation,0),
      opacity:n(t.opacity != null ? t.opacity : obj.opacity,1)
    };
  }

  function cloneTransform(t){
    var W = getWick();
    if(W && W.Transformation){
      try{return new W.Transformation(t || {});}catch(e1){}
      try{return new W.Transformation();}catch(e2){}
    }
    return clonePlain(t || {});
  }

  function applyTransform(obj, values){
    if(!obj) return;
    var current = {};
    try{current = obj.transformation || {};}catch(e0){}
    var t = cloneTransform(current);
    t.x = values.x; t.y = values.y; t.scaleX = values.scaleX; t.scaleY = values.scaleY;
    t.rotation = values.rotation; t.opacity = values.opacity;
    try{obj.transformation = t;}catch(e1){}
    var keys = ['x','y','scaleX','scaleY','rotation','opacity'];
    for(var i=0;i<keys.length;i++){
      try{obj[keys[i]] = values[keys[i]];}catch(e2){}
    }
  }

  function createPathFromJSON(json, project){
    var W = getWick();
    if(!W || !W.Path) return null;
    var p = null;
    try{p = new W.Path({json:clonePlain(json), project:project});}catch(e1){p=null;}
    if(!p){try{p = new W.Path({json:clonePlain(json)});}catch(e2){p=null;}}
    if(!p){try{p = new W.Path(clonePlain(json));}catch(e3){p=null;}}
    try{if(p && p.updateJSON) p.updateJSON();}catch(e4){}
    return p;
  }

  function findClipInSelection(project){
    var arr = selectedObjects(project);
    for(var i=0;i<arr.length;i++){
      var o=arr[i];
      var c=String(o && (o.classname||o.className||(o.constructor&&o.constructor.name)) || '').toLowerCase();
      if(o && (o.timeline || c.indexOf('clip')>=0)) return o;
    }
    return null;
  }

  function firstLayer(clip){
    try{
      if(clip && clip.timeline){
        var ls = clip.timeline.layers || clip.timeline._children || [];
        if(ls.length) return ls[0];
      }
    }catch(e){}
    return null;
  }

  function frameList(layer){
    try{return (layer.frames || layer._children || []).slice();}catch(e){return [];}
  }

  function removeFrame(layer, frame){
    if(!layer || !frame) return;
    try{if(typeof layer.removeFrame === 'function'){layer.removeFrame(frame);return;}}catch(e1){}
    try{if(typeof frame.remove === 'function'){frame.remove();return;}}catch(e2){}
    try{
      var arr=layer.frames||layer._children;
      var ix=arr.indexOf(frame); if(ix>=0) arr.splice(ix,1);
    }catch(e3){}
  }

  function clearLayerFrames(layer){
    var fs=frameList(layer);
    for(var i=fs.length-1;i>=0;i--) removeFrame(layer,fs[i]);
  }

  function createFrame(layer,start){
    var W=getWick();
    if(!W || !W.Frame) throw new Error('Wick.Frame is unavailable');
    var f=new W.Frame({start:start,length:1});
    if(!layer || typeof layer.addFrame!=='function') throw new Error('Wick layer.addFrame is unavailable');
    layer.addFrame(f);
    return f;
  }

  function addPath(frame,path){
    if(!frame || !path) return false;
    try{if(typeof frame.addPath==='function'){frame.addPath(path);return true;}}catch(e1){}
    try{if(typeof frame.addObject==='function'){frame.addObject(path);return true;}}catch(e2){}
    try{if(Array.isArray(frame.paths)){frame.paths.push(path);return true;}}catch(e3){}
    return false;
  }

  function val(name,d){
    var v=Number(el[name].value); return isFinite(v)?v:d;
  }
  function cfg(){
    return {
      count:Math.max(1,Math.min(300,Math.round(val('count',30)))),
      duration:Math.max(1,Math.min(300,Math.round(val('duration',48)))),
      lifetime:Math.max(1,Math.min(300,Math.round(val('lifetime',24)))),
      direction:val('direction',-90), spread:Math.max(0,val('spread',60)),
      emitterWidth:Math.max(0,val('emitterWidth',0)), emitterHeight:Math.max(0,val('emitterHeight',0)),
      seed:Math.max(0,Math.round(val('seed',12345)))>>>0,
      speed:val('speed',8), speedVar:Math.max(0,val('speedVar',3)),
      gravityX:val('gravityX',0), gravityY:val('gravityY',0.35), drag:Math.max(0,Math.min(.99,val('drag',0))),
      scaleStart:Math.max(0,val('scaleStart',100))/100, scaleEnd:Math.max(0,val('scaleEnd',35))/100,
      scaleVar:Math.max(0,val('scaleVar',20))/100,
      opacityStart:Math.max(0,Math.min(100,val('opacityStart',100)))/100,
      opacityEnd:Math.max(0,Math.min(100,val('opacityEnd',0)))/100,
      rotationStart:val('rotationStart',0), spin:val('spin',4), randomRotation:!!el.randomRotation.checked,
      stagger:!!el.stagger.checked, loopBirths:!!el.loopBirths.checked
    };
  }

  function mulberry32(a){
    return function(){
      a|=0;a=a+0x6D2B79F5|0;var t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;
    };
  }

  function particleTable(c){
    var r=mulberry32(c.seed || 1), out=[];
    for(var i=0;i<c.count;i++){
      var birth=0;
      if(c.stagger) birth=Math.floor((i/Math.max(1,c.count))*c.duration);
      if(c.loopBirths && c.lifetime<c.duration) birth=Math.floor(r()*c.duration);
      var ang=(c.direction + (r()-.5)*c.spread)*Math.PI/180;
      var sp=c.speed + (r()-.5)*2*c.speedVar;
      var sv=1 + (r()-.5)*2*c.scaleVar;
      out.push({
        birth:birth,
        vx:Math.cos(ang)*sp,
        vy:Math.sin(ang)*sp,
        ox:(r()-.5)*c.emitterWidth,
        oy:(r()-.5)*c.emitterHeight,
        rot:c.rotationStart + (c.randomRotation?r()*360:0),
        spin:c.spin,
        scaleVar:Math.max(0,sv),
        sourceIndex:Math.floor(r()*Math.max(1,sourceJSON.length))
      });
    }
    return out;
  }

  function stateAt(p,age,c){
    var x=p.ox,y=p.oy,vx=p.vx,vy=p.vy;
    for(var k=0;k<age;k++){
      x+=vx; y+=vy;
      vx+=c.gravityX; vy+=c.gravityY;
      vx*=1-c.drag; vy*=1-c.drag;
    }
    var q=Math.max(0,Math.min(1,age/Math.max(1,c.lifetime-1)));
    var sc=(c.scaleStart+(c.scaleEnd-c.scaleStart)*q)*p.scaleVar;
    var op=c.opacityStart+(c.opacityEnd-c.opacityStart)*q;
    return {x:x,y:y,scale:sc,opacity:op,rotation:p.rot+p.spin*age};
  }

  function requestRedraw(project){
    try{if(project && project.view && project.view.render) project.view.render();}catch(e1){}
    try{var ed=getEditor();if(ed && ed.setState) ed.setState({project:project});}catch(e2){}
  }

  function writeManagedMetadata(clip,c){
    var payload={tool:'LukeTools.Particles',version:1,config:c,sourceJSON:sourceJSON};
    var text='// LUKE_TOOLS_PARTICLES_V1\nthis._lukeParticles = '+JSON.stringify(payload)+';\n';
    try{clip.defaultScript=text;return true;}catch(e1){}
    try{if(clip.scripts){clip.scripts.default=text;return true;}}catch(e2){}
    return false;
  }

  async function bindSelection(){
    if(busy) return;
    var project=getProject();
    if(!project){setStatus('Could not reach the Wick project.',true);return;}
    var selected=selectedObjects(project).filter(isPath);
    if(!selected.length){setStatus('Select one or more native Wick vector paths, then try again.',true);return;}

    busy=true;
    try{
      sourceJSON=[]; sourceBaseTransforms=[];
      for(var i=0;i<selected.length;i++){
        var j=exportObjectJSON(selected[i]);
        if(j){sourceJSON.push(j);sourceBaseTransforms.push(readTransform(selected[i]));}
      }
      if(!sourceJSON.length) throw new Error('The selected paths could not be serialised.');

      if(!project.createClipFromSelection) throw new Error('Wick createClipFromSelection is unavailable.');
      project.createClipFromSelection({identifier:'ParticleSystem',type:'Clip'});
      managed=findClipInSelection(project);
      if(!managed) throw new Error('ParticleSystem clip was created but could not be reselected.');
      try{managed.identifier='ParticleSystem';}catch(e0){}
      internalLayer=firstLayer(managed);
      if(!internalLayer) throw new Error('ParticleSystem has no editable internal layer.');

      await regenerate();
      setStatus('ParticleSystem bound. Change any control to regenerate native Wick frames.');
    }catch(err){
      setStatus('Bind failed: '+(err&&err.message?err.message:String(err)),true);
    }finally{busy=false;}
  }

  async function regenerate(){
    if(!managed || !sourceJSON.length){setStatus('Bind selected curves first.',true);return;}
    if(busy) return;
    busy=true;
    var project=getProject();
    try{
      var c=cfg();
      var particles=particleTable(c);
      internalLayer=internalLayer||firstLayer(managed);
      if(!internalLayer) throw new Error('Managed clip layer is unavailable.');
      clearLayerFrames(internalLayer);

      for(var frameNo=0;frameNo<c.duration;frameNo++){
        var frame=createFrame(internalLayer,frameNo);
        for(var pi=0;pi<particles.length;pi++){
          var p=particles[pi],age=frameNo-p.birth;
          if(age<0 || age>=c.lifetime) continue;
          var idx=p.sourceIndex%sourceJSON.length;
          var inst=createPathFromJSON(sourceJSON[idx],project);
          if(!inst) continue;
          var base=sourceBaseTransforms[idx]||{x:0,y:0,scaleX:1,scaleY:1,rotation:0,opacity:1};
          var s=stateAt(p,age,c);
          applyTransform(inst,{
            x:base.x+s.x,
            y:base.y+s.y,
            scaleX:base.scaleX*s.scale,
            scaleY:base.scaleY*s.scale,
            rotation:base.rotation+s.rotation,
            opacity:Math.max(0,Math.min(1,base.opacity*s.opacity))
          });
          addPath(frame,inst);
        }
      }
      writeManagedMetadata(managed,c);
      requestRedraw(project);
      setStatus('Generated '+c.duration+' frames · '+c.count+' particles · seed '+c.seed+'.');
    }catch(err){
      setStatus('Regeneration failed: '+(err&&err.message?err.message:String(err)),true);
    }finally{busy=false;}
  }

  function scheduleRegenerate(){
    clearTimeout(debounceTimer);
    debounceTimer=setTimeout(function(){if(managed) regenerate();},180);
  }

  ids.forEach(function(id){
    el[id].addEventListener(el[id].type==='checkbox'?'change':'input',scheduleRegenerate);
  });
  el.bindBtn.addEventListener('click',bindSelection);
  el.rebindBtn.addEventListener('click',bindSelection);
  el.refreshBtn.addEventListener('click',regenerate);
  el.randomSeedBtn.addEventListener('click',function(){el.seed.value=String(Math.floor(Math.random()*2147483647));scheduleRegenerate();});

})();
</script>
