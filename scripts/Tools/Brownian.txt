<script>
(function(){
'use strict';
var currentInfo=null;
function getBridge(){
  try{if(window.LukeToolsBridge)return window.LukeToolsBridge;}catch(e){}
  try{if(window.parent&&window.parent.LukeToolsBridge)return window.parent.LukeToolsBridge;}catch(e){}
  try{if(window.top&&window.top.LukeToolsBridge)return window.top.LukeToolsBridge;}catch(e){}
  return null;
}
function status(msg,kind){var e=document.getElementById('status');e.className='status'+(kind?' '+kind:'');e.textContent=msg;}
function num(id,d){var v=Number(document.getElementById(id).value);return isFinite(v)?v:d;}
function config(){return{
  forceX:Math.max(0,num('forceX',0.8)),forceY:Math.max(0,num('forceY',0.8)),
  inertia:Math.max(0,Math.min(.999,num('inertia',.88))),returnForce:Math.max(0,Math.min(1,num('returnForce',.025))),
  driftX:num('driftX',0),driftY:num('driftY',0),maxX:Math.max(0,num('maxX',80)),maxY:Math.max(0,num('maxY',80)),
  rotation:Math.max(0,num('rotation',0)),seed:(Math.max(0,Math.floor(num('seed',12345)))>>>0)
};}
function parseConfig(s){
  var m=String(s||'').match(/\/\/ Brownian CONFIG (\{[^\n]*\})/);
  if(!m)return null;try{return JSON.parse(m[1]);}catch(e){return null;}
}
function applyConfig(c){if(!c)return;Object.keys(c).forEach(function(k){var e=document.getElementById(k);if(e)e.value=String(c[k]);});}
function refresh(){
  var b=getBridge();
  if(!b||typeof b.getSelectionInfo!=='function'){currentInfo=null;document.getElementById('selLabel').textContent='No bridge';status('LukeToolsBridge is unavailable.','bad');return;}
  try{
    var i=b.getSelectionInfo();
    if(i&&i.uuid){
      currentInfo=i;document.getElementById('selLabel').textContent=i.identifier||i.name||i.uuid;
      var s='';try{if(typeof b.getUpdateScriptByInfo==='function')s=b.getUpdateScriptByInfo(i)||'';}catch(e1){}
      if(!s){try{if(typeof b.getUpdateScriptOnSelection==='function')s=b.getUpdateScriptOnSelection()||'';}catch(e2){}}
      var c=parseConfig(s);if(c){applyConfig(c);status('Loaded existing Brownian settings.','warn');}else status('Clip selected. Configure Brownian and click Create / Update.');
      return;
    }
  }catch(e){}
  currentInfo=null;document.getElementById('selLabel').textContent='No Clip Chosen';status('Select a clip on the Wick stage, then click Refresh.','bad');
}
function buildScript(c){
  var j=JSON.stringify(c);
  return [
    '// Brownian BEGIN',
    '// Brownian CONFIG '+j,
    '(function(){',
    '  var C='+j+';',
    '  var S=this.__ltBrownian;',
    '  if(!S || S.version!==1 || S.seed!==C.seed){',
    '    S=this.__ltBrownian={version:1,seed:C.seed,state:(C.seed||1)>>>0,vx:0,vy:0,vr:0,homeX:Number(this.x)||0,homeY:Number(this.y)||0,homeR:Number(this.rotation)||0};',
    '  }',
    '  function rnd(){',
    '    var a=S.state|0;a=a+0x6D2B79F5|0;var t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;S.state=a;return((t^t>>>14)>>>0)/4294967296;',
    '  }',
    '  var x=Number(this.x)||0,y=Number(this.y)||0,r=Number(this.rotation)||0;',
    '  S.vx=(S.vx+(rnd()*2-1)*C.forceX+C.driftX+(S.homeX-x)*C.returnForce)*C.inertia;',
    '  S.vy=(S.vy+(rnd()*2-1)*C.forceY+C.driftY+(S.homeY-y)*C.returnForce)*C.inertia;',
    '  x+=S.vx;y+=S.vy;',
    '  if(C.maxX>0){var dx=x-S.homeX;if(dx>C.maxX){x=S.homeX+C.maxX;S.vx*=-0.35;}else if(dx<-C.maxX){x=S.homeX-C.maxX;S.vx*=-0.35;}}',
    '  if(C.maxY>0){var dy=y-S.homeY;if(dy>C.maxY){y=S.homeY+C.maxY;S.vy*=-0.35;}else if(dy<-C.maxY){y=S.homeY-C.maxY;S.vy*=-0.35;}}',
    '  if(C.rotation>0){S.vr=(S.vr+(rnd()*2-1)*C.rotation+(S.homeR-r)*C.returnForce)*C.inertia;r+=S.vr;}',
    '  this.x=x;this.y=y;if(C.rotation>0)this.rotation=r;',
    '}).call(this);',
    '// Brownian END'
  ].join('\n');
}
function createOrUpdate(){
  if(!currentInfo){refresh();if(!currentInfo)return;}
  var b=getBridge();if(!b){status('LukeToolsBridge is unavailable.','bad');return;}
  var existing='';
  try{if(typeof b.getUpdateScriptByInfo==='function')existing=b.getUpdateScriptByInfo(currentInfo)||'';}catch(e1){}
  if(!existing){try{if(typeof b.getUpdateScriptOnSelection==='function')existing=b.getUpdateScriptOnSelection()||'';}catch(e2){}}
  existing=String(existing||'').replace(/\/\/ Brownian BEGIN[\s\S]*?\/\/ Brownian END\n?/g,'').trim();
  var brown=buildScript(config());
  var combined=existing?(existing+'\n\n'+brown):brown;
  try{
    var res=null;
    if(typeof b.setUpdateScriptByInfo==='function')res=b.setUpdateScriptByInfo(currentInfo,combined);
    if(!res||res.ok===false){if(typeof b.setUpdateScriptOnSelection==='function')res=b.setUpdateScriptOnSelection(combined);}
    if(res&&res.ok===false)throw new Error(res.reason||res.error||'Bridge rejected the script');
    status('Brownian behaviour created/updated on '+(currentInfo.identifier||currentInfo.name||'selected clip')+'.','good');
  }catch(err){status('Could not write Brownian behaviour: '+(err&&err.message?err.message:String(err)),'bad');}
}
document.getElementById('refreshBtn').addEventListener('click',refresh);
document.getElementById('createBtn').addEventListener('click',createOrUpdate);
refresh();
})();
