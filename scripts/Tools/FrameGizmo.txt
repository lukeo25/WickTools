<script>
var F_Screen = 0;
(function(){
  "use strict";

  var state={clip:null,stops:[],signature:"",lastUUID:"",lastFrame:0};
  var $=function(id){return document.getElementById(id);};

  function bridge(){
    try{return window.LukeToolsBridge||(window.parent&&window.parent.LukeToolsBridge)||(window.top&&window.top.LukeToolsBridge)||null;}catch(e){return null;}
  }
  function editor(){
    var b=bridge();
    try{if(b&&typeof b.getEditor==="function")return b.getEditor();}catch(e0){}
    try{if(window.parent&&window.parent.wickEditor)return window.parent.wickEditor;}catch(e1){}
    try{if(window.parent&&window.parent.WickEditor&&window.parent.WickEditor.editor)return window.parent.WickEditor.editor;}catch(e2){}
    try{if(window.parent&&window.parent.editor)return window.parent.editor;}catch(e3){}
    return null;
  }
  function project(){
    var b=bridge(),ed=editor();
    try{if(b&&typeof b.getProject==="function")return b.getProject();}catch(e0){}
    try{return ed&&ed.project?ed.project:null;}catch(e1){return null;}
  }
  function uniquePush(list,obj){if(obj&&list.indexOf(obj)<0)list.push(obj);}
  function selectedObjects(){
    var b=bridge(),p=project(),ed=editor(),out=[];
    try{if(b&&typeof b.getSelectedObjects==="function"){var a=b.getSelectedObjects();if(a&&a.length)Array.prototype.slice.call(a).forEach(function(o){uniquePush(out,o);});}}catch(e0){}
    var sources=[];
    try{sources.push(p&&elf(p,"selection"));}catch(e1){}
    try{sources.push(ed&&elf(ed,"selection"));}catch(e2){}
    sources.forEach(function(sel){
      if(!sel)return;
      try{if(typeof sel.getSelectedObjects==="function"){var a=sel.getSelectedObjects();if(a&&a.length)Array.prototype.slice.call(a).forEach(function(o){uniquePush(out,o);});}}catch(e3){}
      try{if(sel.objects&&sel.objects.length)Array.prototype.slice.call(sel.objects).forEach(function(o){uniquePush(out,o);});}catch(e4){}
      try{if(sel.selectedObjects&&sel.selectedObjects.length)Array.prototype.slice.call(sel.selectedObjects).forEach(function(o){uniquePush(out,o);});}catch(e5){}
      try{if(typeof sel.getSelectedObject==="function")uniquePush(out,sel.getSelectedObject());}catch(e6){}
      try{if(typeof sel.getPrimarySelectedObject==="function")uniquePush(out,sel.getPrimarySelectedObject());}catch(e7){}
    });
    try{if(!out.length&&ed&&ed.selectedObjects&&ed.selectedObjects.length)Array.prototype.slice.call(ed.selectedObjects).forEach(function(o){uniquePush(out,o);});}catch(e8){}
    return out;
  }
  function elf(obj,key){return obj?obj[key]:null;}
  function isClip(obj){
    if(!obj)return false;
    try{var W=(window.parent&&window.parent.Wick)||window.Wick;if(W&&W.Clip&&obj instanceof W.Clip)return true;}catch(e0){}
    try{if(obj.classname==="Clip"&&obj.timeline)return true;}catch(e1){}
    try{if(obj.timeline&&obj.transformation&&!obj.json)return true;}catch(e2){}
    return false;
  }
  function selectedClip(){
    var arr=selectedObjects().filter(isClip);
    return arr.length===1?arr[0]:null;
  }
  function clipName(clip){
    if(!clip)return "Select one Clip";
    try{return String(clip.identifier||clip.name||clip.uuid||"Selected Clip");}catch(e){return "Selected Clip";}
  }
  function currentFrame(clip){
    var n=1;
    try{n=Number(clip.singleFrameNumber||clip.timeline.playheadPosition||1);}catch(e){}
    return Math.max(1,Math.round(isFinite(n)?n:1));
  }
  function frameStart(frame){
    var n=1;try{n=Number(frame.start||frame.startFrame||1);}catch(e){}
    return Math.max(1,Math.round(isFinite(n)?n:1));
  }
  function frameLabel(frame){
    try{return String(frame.identifier||frame.name||"").trim();}catch(e){return "";}
  }
  function collectStops(clip){
    var map={},layers=[];
    try{layers=clip.timeline&&clip.timeline.layers?clip.timeline.layers:[];}catch(e0){}
    Array.prototype.slice.call(layers||[]).forEach(function(layer){
      var frames=[];
      try{frames=layer&&layer.frames?layer.frames:[];}catch(e1){frames=[];}
      Array.prototype.slice.call(frames||[]).forEach(function(frame){
        if(!frame)return;
        var n=frameStart(frame),label=frameLabel(frame);
        if(!map[n])map[n]={frame:n,label:label||("Frame "+n),named:!!label,source:frame};
        else if(label&&!map[n].named){map[n].label=label;map[n].named=true;map[n].source=frame;}
      });
    });
    if(!map[1])map[1]={frame:1,label:"Frame 1",named:false,source:null};
    return Object.keys(map).map(function(k){return map[k];}).sort(function(a,b){return a.frame-b.frame;});
  }
  function mode(clip){
    if(!clip)return "Waiting";
    try{if(clip.animationType==="single")return "Single";if(clip.isSynced)return "Follow";return clip.animationType==="playOnce"?"Play once":"Loop";}catch(e){return "Clip";}
  }
  function markDirty(label){
    var b=bridge(),ed=editor(),p=project();
    try{if(b&&typeof b.markDirty==="function")b.markDirty();}catch(e0){}
    try{if(p)p.unsavedChanges=true;}catch(e1){}
    try{if(p&&typeof p.markAsModified==="function")p.markAsModified();}catch(e2){}
    try{if(ed&&typeof ed.projectDidChange==="function")ed.projectDidChange({actionName:label||"LukeTools Gizmo"});}catch(e3){try{if(ed&&typeof ed.projectDidChange==="function")ed.projectDidChange();}catch(e4){}}
    try{if(ed&&ed.canvas&&typeof ed.canvas.render==="function")ed.canvas.render();}catch(e5){}
    try{if(ed&&typeof ed.syncInterfaces==="function")ed.syncInterfaces();}catch(e6){}
    try{if(ed&&typeof ed.updateUI==="function")ed.updateUI();}catch(e7){}
  }
  function status(text,type){var box=$("status");box.textContent=text;box.className=type||"";}
  function setFrame(frame){
    var clip=state.clip;if(!clip)return status("Select one Clip first.","error");
    var n=Math.max(1,Math.round(Number(frame)||1));
    try{clip.isSynced=false;}catch(e0){}
    try{clip.animationType="single";}catch(e1){}
    try{clip.singleFrameNumber=n;}catch(e2){}
    try{if(clip.timeline)clip.timeline.playheadPosition=n;}catch(e3){}
    markDirty("Set Gizmo frame "+n);
    status("Showing "+clipName(clip)+" frame "+n+".","ok");
    state.signature="";render();
  }
  function setMode(next){
    var clip=state.clip;if(!clip)return status("Select one Clip first.","error");
    try{
      if(next==="single"){
        var n=currentFrame(clip);clip.isSynced=false;clip.animationType="single";clip.singleFrameNumber=n;
      }else if(next==="follow"){
        clip.animationType="loop";clip.isSynced=true;if(typeof clip.applySyncPosition==="function")clip.applySyncPosition();
      }else{
        clip.isSynced=false;clip.animationType="loop";if(clip.timeline)clip.timeline.playheadPosition=1;
      }
    }catch(e){return status("Could not change Clip playback: "+e.message,"error");}
    markDirty("Set Gizmo playback");state.signature="";render();status(next==="follow"?"Clip now follows its parent timeline.":next==="loop"?"Ordinary Clip looping restored.":"Clip will hold its selected internal frame.","ok");
  }
  function matchingFrameForLabel(clip,n){
    var layers=[];try{layers=clip.timeline.layers||[];}catch(e0){}
    var fallback=null;
    for(var li=0;li<layers.length;li++){
      var frames=[];try{frames=layers[li].frames||[];}catch(e1){}
      for(var fi=0;fi<frames.length;fi++){
        var frame=frames[fi];if(!frame)continue;
        if(frameStart(frame)===n){if(frameLabel(frame))return frame;if(!fallback)fallback=frame;}
      }
    }
    return fallback;
  }
  function validIdentifier(value){
    var s=String(value||"").trim().replace(/[^A-Za-z0-9_$]+/g,"_").replace(/^_+|_+$/g,"");
    if(!s)return "";if(!/^[A-Za-z_$]/.test(s))s="frame_"+s;return s;
  }
  function saveLabel(){
    var clip=state.clip;if(!clip)return status("Select one Clip first.","error");
    var n=currentFrame(clip),value=validIdentifier($("frameLabel").value);
    if(!value)return status("Enter a frame label first.","error");
    var frame=matchingFrameForLabel(clip,n);
    if(!frame)return status("No internal key frame starts at frame "+n+".","error");
    try{frame.identifier=value;}catch(e0){}
    try{frame.name=value;}catch(e1){}
    markDirty("Name Gizmo frame");state.signature="";render();status("Frame "+n+" labelled "+value+".","ok");
  }
  function neighbouring(direction){
    if(!state.clip||!state.stops.length)return;
    var n=currentFrame(state.clip),index=0;
    for(var i=0;i<state.stops.length;i++){if(state.stops[i].frame<=n)index=i;}
    index=Math.max(0,Math.min(state.stops.length-1,index+direction));setFrame(state.stops[index].frame);
  }
  function stopAngle(index,count){return count<=1?-90:-155+(130*index/(count-1));}
  function renderDial(stops,active){
    var dial=$("dial");dial.innerHTML="";
    if(!state.clip){dial.innerHTML='<div class="empty">Select one Clip instance on the Wick stage.<br>The Clip\'s internal key frames will appear here.</div>';return;}
    if(!stops.length){dial.innerHTML='<div class="empty">This Clip has no internal key frames.</div>';return;}
    var disc=document.createElement("div");disc.className="disc";dial.appendChild(disc);
    var shown=stops.length>12?stops.slice(0,12):stops;
    var activeIndex=0;
    shown.forEach(function(stop,i){if(stop.frame<=active)activeIndex=i;});
    var activeStopFrame=shown[activeIndex].frame;
    var angle=stopAngle(activeIndex,shown.length),arm=document.createElement("div");arm.className="arm";arm.style.transform="rotate("+angle+"deg)";dial.appendChild(arm);
    var pivot=document.createElement("div");pivot.className="pivot";dial.appendChild(pivot);
    shown.forEach(function(stop,i){
      var a=stopAngle(i,shown.length)*Math.PI/180,r=102,cx=142,cy=104;
      var button=document.createElement("button");button.type="button";button.className="stop"+(stop.frame===activeStopFrame?" active":"");
      button.style.left=Math.round(cx+Math.cos(a)*r-27)+"px";button.style.top=Math.round(cy+Math.sin(a)*r-27)+"px";
      var text=stop.named?stop.label:("Fr "+stop.frame);if(text.length>11)text=text.slice(0,10)+"…";
      button.innerHTML=text+'<small>Fr '+stop.frame+'</small>';button.title=(stop.named?stop.label+" — ":"")+"Frame "+stop.frame;
      button.addEventListener("click",function(){setFrame(stop.frame);});dial.appendChild(button);
    });
    if(stops.length>shown.length){var note=document.createElement("div");note.className="muted";note.style.cssText="position:absolute;left:0;right:0;bottom:0;text-align:center";note.textContent="First 12 of "+stops.length+" key frames shown";dial.appendChild(note);}
  }
  function render(force){
    var clip=selectedClip();
    if(clip)state.clip=clip;else state.clip=null;
    state.stops=state.clip?collectStops(state.clip):[];
    var uuid="";try{uuid=state.clip&&state.clip.uuid||"";}catch(e0){}
    var active=state.clip?currentFrame(state.clip):0,m=mode(state.clip);
    var sig=[uuid,active,m,state.stops.map(function(s){return s.frame+":"+s.label;}).join("|")].join("~");
    if(!force&&sig===state.signature)return;state.signature=sig;
    $("clipName").textContent=clipName(state.clip);$("modeBadge").textContent=m;
    $("frameReadout").textContent=state.clip?("Frame "+active+" of "+Math.max(1,Number(state.clip.timeline&&state.clip.timeline.length)||1)):"No Clip selected";
    renderDial(state.stops,active);
    var stop=null;state.stops.forEach(function(candidate){if(candidate.frame<=active)stop=candidate;});$("frameLabel").value=stop&&stop.named?stop.label:"";
    ["previous","next","single","follow","loop","saveLabel"].forEach(function(id){$(id).disabled=!state.clip;});
    $("single").classList.toggle("active",m==="Single");$("follow").classList.toggle("active",m==="Follow");$("loop").classList.toggle("active",m==="Loop");
    if(!state.clip&&uuid!==state.lastUUID)status("Select exactly one Clip instance on the Wick stage.","error");
    state.lastUUID=uuid;state.lastFrame=active;
  }

  $("refresh").addEventListener("click",function(){state.signature="";render(true);});
  $("previous").addEventListener("click",function(){neighbouring(-1);});
  $("next").addEventListener("click",function(){neighbouring(1);});
  $("single").addEventListener("click",function(){setMode("single");});
  $("follow").addEventListener("click",function(){setMode("follow");});
  $("loop").addEventListener("click",function(){setMode("loop");});
  $("saveLabel").addEventListener("click",saveLabel);
  $("frameLabel").addEventListener("keydown",function(event){if(event.key==="Enter")saveLabel();});
  render(true);setInterval(function(){render(false);},300);
})();
</script>
