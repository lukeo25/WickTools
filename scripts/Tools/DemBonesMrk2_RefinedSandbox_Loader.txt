  function fail(msg){
    var e=document.getElementById("status");
    if(e)e.textContent="DemBones Refined Sandbox failed to load: "+msg;
  }

    var i=text.indexOf(oldText);
    if(i<0) throw new Error("Missing patch anchor: "+label);

  }

  function patch(html){

    // -------------------------------------------------------------------------

    // -------------------------------------------------------------------------

      /<input type="file" id="fileImageLoader" accept="\.svg,image\/svg\+xml" style="display:none;">/g,

      "\n          var pngReader = new FileReader();"+
      "\n          pngReader.onerror = function(){ reject(new Error('Could not read PNG file.')); };"+

      "\n            try {"+
      "\n              var pngDataUrl = String(pngReader.result || '');"+
      "\n              var pngImage = new Image();"+
      "\n              pngImage.onload = function(){"+
      "\n                resolve({"+
      "\n                  dataUrl: pngDataUrl,"+
      "\n                  image: pngImage,"+
      "\n                  fileName: String(file.name || 'image.png'),"+
      "\n                  svgRaw: '',"+
      "\n                  svgNative: null,"+
      "\n                  width: Math.max(1,pngImage.naturalWidth || pngImage.width || 1),"+
