function viewportDistancePointToSegment(px,py,a,b){
  var vx=b.x-a.x,vy=b.y-a.y,wx=px-a.x,wy=py-a.y,den=vx*vx+vy*vy,t=den>0?(wx*vx+wy*vy)/den:0;
  t=Math.max(0,Math.min(1,t));var x=a.x+vx*t,y=a.y+vy*t,dx=px-x,dy=py-y;return Math.sqrt(dx*dx+dy*dy);
}

function enforcePoseConnections(part,pose){
  var byId={};(part.bones||[]).forEach(function(b){byId[String(b.id)]=b;});
  var guard=0,changed=true;
  while(changed&&guard++<20){
    changed=false;
    (part.bones||[]).forEach(function(b){
      if(!b.parentId)return;
      var parent=pose[String(b.parentId)],child=pose[String(b.id)];
      if(!parent||!child)return;
      var dx=parent.tail.x-child.head.x,dy=parent.tail.y-child.head.y;
      if(Math.abs(dx)>0.0001||Math.abs(dy)>0.0001){
        child.head.x+=dx;child.head.y+=dy;child.tail.x+=dx;child.tail.y+=dy;changed=true;
      }
    });
  }
  return pose;
}

function collectPointsFromPaperJson(json,out){
  out=out||[];
  function addPoint(v){
    if(Array.isArray(v)&&v.length>=2&&isFinite(Number(v[0]))&&isFinite(Number(v[1])))out.push({x:Number(v[0]),y:Number(v[1])});
    else if(v&&typeof v==="object"&&isFinite(Number(v.x))&&isFinite(Number(v.y)))out.push({x:Number(v.x),y:Number(v.y)});
  }
  function walk(node,key){
    if(node==null)return;
    if(Array.isArray(node)){
      if(key==="segments"){
        node.forEach(function(seg){
          if(Array.isArray(seg)&&seg.length){addPoint(seg[0]);}
          else if(seg&&typeof seg==="object"&&seg.point)addPoint(seg.point);
        });
      }
      node.forEach(function(v){walk(v,"");});
      return;
    }
    if(typeof node!=="object")return;
    if(Array.isArray(node.segments))walk(node.segments,"segments");
    Object.keys(node).forEach(function(k){if(k!=="segments")walk(node[k],k);});
  }
  walk(json,"");
  return out;
}

  });
  return unique;
}
function autoSizeAllEnvelopesNative(){

    s.count++;
  });

    if(!s||!s.count){

      });
    }

  });
