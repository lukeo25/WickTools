<script>
(function(){
"use strict";

const SVGNS = "http://www.w3.org/2000/svg";
const svgRoot = document.getElementById("svgRoot");
const drawLayer = document.getElementById("drawLayer");
const pageRect = document.getElementById("pageRect");
const stage = document.getElementById("stage");
const canvasWrap = document.getElementById("canvasWrap");
const selOverlay = document.getElementById("selOverlay");

let pageW = 1280, pageH = 800;
let zoom = 1, panX = 60, panY = 40;
let currentTool = "select";
let selected = []; // array of SVG elements
let idCounter = 1;
let history = [], historyIndex = -1;
let isPanning = false, spaceDown = false;
let nodeEditTarget = null; // path element being node-edited
let nodeEditIndex = -1;

// ---------- Tool definitions ----------
const TOOLS = [
  {id:"select", label:"Select (V) — double-click a shape to edit its curves", icon:"M4 3l6 15 2-6 6-2z"},
  {id:"node", label:"Edit Nodes (N)", icon:"M4 4h4v4H4zM16 16h4v4h-4zM8 8l8 8"},
  {id:"rect", label:"Rectangle (R)", icon:"M4 5h16v14H4z"},
  {id:"ellipse", label:"Ellipse (E)", icon:"circle:12,12,9"},
  {id:"line", label:"Line (L)", icon:"M4 20L20 4"},
  {id:"pen", label:"Pen / Path (P)", icon:"M4 20l4-1 11-11-3-3L5 16z"},
  {id:"texture", label:"Texture Pen (B)", icon:"M3 21l4-1 12-12-3-3L4 17zM14 5l3 3"},
  {id:"shapes", label:"LukeTools Shapes (S)", icon:"M4 4h7v7H4zM14 4h6v6h-6zM5 15l4-3 4 3-4 5zM16 14a3 3 0 1 0 0 6a3 3 0 0 0 0-6z"},
  {id:"text", label:"Text (T)", icon:"M5 5h14M12 5v14"},
  {id:"pan", label:"Pan (H / Space)", icon:"M12 3v18M3 12h18"}
];

const toolbar = document.getElementById("toolbar");
TOOLS.forEach(t=>{
  const b = document.createElement("button");
  b.className = "tool-btn" + (t.id==="select"?" active":"");
  b.title = t.label;
  b.dataset.tool = t.id;
  b.innerHTML = `<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">${
    t.icon.startsWith("circle:") ? (()=>{const [cx,cy,r]=t.icon.slice(7).split(",");return `<circle cx="${cx}" cy="${cy}" r="${r}"/>`;})() : `<path d="${t.icon}"/>`
  }</svg>`;
  b.addEventListener("click", ()=>setTool(t.id));
  toolbar.appendChild(b);
});

function setTool(id){
  if(id === "shapes"){
    openShapesDrawer();
    return;
  }
  currentTool = id;
  document.querySelectorAll(".tool-btn").forEach(b=>b.classList.toggle("active", b.dataset.tool===id));
  document.getElementById("statusTool").textContent = TOOLS.find(t=>t.id===id).label.split(" (")[0];
  document.getElementById("texturePanel").classList.toggle("visible", id==="texture");
  if(id !== "node"){ exitNodeEdit(); }
  clearSelection();
  updateCursor();
}
function updateCursor(){
  canvasWrap.style.cursor = currentTool==="pan" ? "grab" : (currentTool==="select"?"default":"crosshair");
}

// ---------- Coordinate helpers ----------
function applyStageTransform(){
  stage.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`;
  document.getElementById("zoomLabel").textContent = Math.round(zoom*100)+"%";
  syncOverlaySize();
}
function screenToSvg(clientX, clientY){
  const r = canvasWrap.getBoundingClientRect();
  const x = (clientX - r.left - panX) / zoom;
  const y = (clientY - r.top - panY) / zoom;
  return {x, y};
}
function syncOverlaySize(){
  const r = canvasWrap.getBoundingClientRect();
  selOverlay.setAttribute("width", r.width);
  selOverlay.setAttribute("height", r.height);
  selOverlay.style.width = r.width+"px";
  selOverlay.style.height = r.height+"px";
}
function svgToScreen(x,y){
  const r = canvasWrap.getBoundingClientRect();
  return {x: x*zoom+panX, y: y*zoom+panY};
}

// ---------- History (undo/redo) ----------
function snapshot(){
  return drawLayer.innerHTML;
}
function pushHistory(){
  history = history.slice(0, historyIndex+1);
  history.push(snapshot());
  if(history.length>100) history.shift();
  historyIndex = history.length-1;
  updateLayerList();
  updateStatusCount();
}
function undo(){
  if(historyIndex<=0) return;
  historyIndex--;
  drawLayer.innerHTML = history[historyIndex];
  clearSelection();
  updateLayerList();
  updateStatusCount();
}
function redo(){
  if(historyIndex>=history.length-1) return;
  historyIndex++;
  drawLayer.innerHTML = history[historyIndex];
  clearSelection();
  updateLayerList();
  updateStatusCount();
}

// ---------- Style state ----------
function currentStyle(){
  const fillOn = document.getElementById("fillOn").checked;
  const strokeOn = document.getElementById("strokeOn").checked;
  return {
    fill: fillOn ? document.getElementById("fillColor").value : "none",
    stroke: strokeOn ? document.getElementById("strokeColor").value : "none",
    strokeWidth: document.getElementById("strokeWidth").value,
    opacity: document.getElementById("opacity").value/100
  };
}
function applyStyleToEl(el, style){
  el.setAttribute("fill", style.fill);
  el.setAttribute("stroke", style.stroke);
  el.setAttribute("stroke-width", style.strokeWidth);
  el.setAttribute("opacity", style.opacity);
}

// ---------- Shape creation ----------
function newId(){ return "obj"+(idCounter++); }

function createRect(x,y,w,h){
  const el = document.createElementNS(SVGNS,"rect");
  el.setAttribute("x",Math.min(x,x+w)); el.setAttribute("y",Math.min(y,y+h));
  el.setAttribute("width",Math.abs(w)); el.setAttribute("height",Math.abs(h));
  el.id = newId();
  applyStyleToEl(el, currentStyle());
  return el;
}
function createEllipse(x,y,w,h){
  const el = document.createElementNS(SVGNS,"ellipse");
  el.setAttribute("cx", x+w/2); el.setAttribute("cy", y+h/2);
  el.setAttribute("rx", Math.abs(w/2)); el.setAttribute("ry", Math.abs(h/2));
  el.id = newId();
  applyStyleToEl(el, currentStyle());
  return el;
}
function createLine(x1,y1,x2,y2){
  const el = document.createElementNS(SVGNS,"line");
  el.setAttribute("x1",x1); el.setAttribute("y1",y1);
  el.setAttribute("x2",x2); el.setAttribute("y2",y2);
  el.id = newId();
  const st = currentStyle();
  el.setAttribute("stroke", st.stroke==="none" ? "#1a1a1a" : st.stroke);
  el.setAttribute("stroke-width", st.strokeWidth);
  el.setAttribute("opacity", st.opacity);
  el.setAttribute("fill","none");
  return el;
}
function createText(x,y,str){
  const el = document.createElementNS(SVGNS,"text");
  el.setAttribute("x",x); el.setAttribute("y",y);
  el.setAttribute("font-size","32");
  el.setAttribute("font-family","sans-serif");
  el.textContent = str;
  el.id = newId();
  const st = currentStyle();
  el.setAttribute("fill", st.fill==="none" ? "#1a1a1a" : st.fill);
  el.setAttribute("opacity", st.opacity);
  return el;
}
function createPath(dStr){
  const el = document.createElementNS(SVGNS,"path");
  el.setAttribute("d", dStr);
  el.id = newId();
  applyStyleToEl(el, currentStyle());
  return el;
}

// ---------- Selection & overlay ----------
function clearSelection(){
  selected = [];
  renderOverlay();
  updateTransformPanel();
  updateLayerList();
}
function setSelection(els){
  selected = els.filter(Boolean);
  renderOverlay();
  updateTransformPanel();
  updateLayerList();
}
function getBBox(el){
  try { return el.getBBox(); } catch(e){ return {x:0,y:0,width:0,height:0}; }
}
function renderOverlay(){
  selOverlay.innerHTML = "";
  if(!selected.length) return;
  if(currentTool==="node" && nodeEditTarget){
    renderNodeHandles();
    return;
  }
  selected.forEach(el=>{
    const bb = getBBox(el);
    const p1 = svgToScreen(bb.x, bb.y);
    const p2 = svgToScreen(bb.x+bb.width, bb.y+bb.height);
    const box = document.createElementNS(SVGNS,"rect");
    box.setAttribute("x",p1.x); box.setAttribute("y",p1.y);
    box.setAttribute("width", p2.x-p1.x); box.setAttribute("height", p2.y-p1.y);
    box.setAttribute("class","sel-box");
    selOverlay.appendChild(box);
    if(selected.length===1){
      const corners = [[p1.x,p1.y],[p2.x,p1.y],[p1.x,p2.y],[p2.x,p2.y],
                        [(p1.x+p2.x)/2,p1.y],[(p1.x+p2.x)/2,p2.y],[p1.x,(p1.y+p2.y)/2],[p2.x,(p1.y+p2.y)/2]];
      corners.forEach((c,i)=>{
        const h = document.createElementNS(SVGNS,"rect");
        h.setAttribute("x",c[0]-4); h.setAttribute("y",c[1]-4);
        h.setAttribute("width",8); h.setAttribute("height",8);
        h.setAttribute("class","handle");
        h.dataset.handle = i;
        selOverlay.appendChild(h);
      });
      const rh = document.createElementNS(SVGNS,"circle");
      rh.setAttribute("cx",(p1.x+p2.x)/2); rh.setAttribute("cy",p1.y-20); rh.setAttribute("r",5);
      rh.setAttribute("class","rot-handle"); rh.dataset.handle="rot";
      selOverlay.appendChild(rh);
      const stem = document.createElementNS(SVGNS,"line");
      stem.setAttribute("x1",(p1.x+p2.x)/2); stem.setAttribute("y1",p1.y);
      stem.setAttribute("x2",(p1.x+p2.x)/2); stem.setAttribute("y2",p1.y-20);
      stem.setAttribute("stroke","#4f8cff"); stem.setAttribute("stroke-width","1");
      selOverlay.appendChild(stem);
    }
  });
  attachHandleEvents();
}

// ---------- Node editing (paths) ----------
function pathToPoints(el){
  // Parses simple M/L/C/Q/Z path data into an editable point list.
  const d = el.getAttribute("d") || "";
  const cmds = d.match(/[MLCQZ][^MLCQZ]*/gi) || [];
  const pts = [];
  let cur = {x:0,y:0};
  cmds.forEach(c=>{
    const type = c[0];
    const nums = c.slice(1).trim().split(/[\s,]+/).filter(s=>s.length).map(Number);
    if(type.toUpperCase()==="M"){ cur={x:nums[0],y:nums[1]}; pts.push({type:"M", x:nums[0], y:nums[1]}); }
    else if(type.toUpperCase()==="L"){ cur={x:nums[0],y:nums[1]}; pts.push({type:"L", x:nums[0], y:nums[1]}); }
    else if(type.toUpperCase()==="C"){ pts.push({type:"C", c1x:nums[0],c1y:nums[1],c2x:nums[2],c2y:nums[3],x:nums[4],y:nums[5]}); cur={x:nums[4],y:nums[5]}; }
    else if(type.toUpperCase()==="Q"){ pts.push({type:"Q", c1x:nums[0],c1y:nums[1],x:nums[2],y:nums[3]}); cur={x:nums[2],y:nums[3]}; }
    else if(type.toUpperCase()==="Z"){ pts.push({type:"Z"}); }
  });
  return pts;
}
function pointsToPath(pts){
  return pts.map(p=>{
    if(p.type==="M") return `M ${p.x} ${p.y}`;
    if(p.type==="L") return `L ${p.x} ${p.y}`;
    if(p.type==="C") return `C ${p.c1x} ${p.c1y} ${p.c2x} ${p.c2y} ${p.x} ${p.y}`;
    if(p.type==="Q") return `Q ${p.c1x} ${p.c1y} ${p.x} ${p.y}`;
    if(p.type==="Z") return "Z";
    return "";
  }).join(" ");
}
function evalSegment(p0, seg, t){
  const mt = 1-t;
  if(seg.type==="L") return {x:p0.x+(seg.x-p0.x)*t, y:p0.y+(seg.y-p0.y)*t};
  if(seg.type==="C") return {
    x: mt*mt*mt*p0.x + 3*mt*mt*t*seg.c1x + 3*mt*t*t*seg.c2x + t*t*t*seg.x,
    y: mt*mt*mt*p0.y + 3*mt*mt*t*seg.c1y + 3*mt*t*t*seg.c2y + t*t*t*seg.y
  };
  if(seg.type==="Q") return {
    x: mt*mt*p0.x + 2*mt*t*seg.c1x + t*t*seg.x,
    y: mt*mt*p0.y + 2*mt*t*seg.c1y + t*t*seg.y
  };
  return p0;
}
function findClosestSegment(pts, point){
  let best = null, anchor = null, subpathStart = null;
  for(let i=0;i<pts.length;i++){
    const seg = pts[i];
    if(seg.type==="M"){ anchor = {x:seg.x,y:seg.y}; subpathStart = anchor; continue; }
    if(seg.type==="Z"){ anchor = subpathStart; continue; }
    const p0 = anchor;
    for(let s=0;s<=16;s++){
      const t = s/16;
      const p = evalSegment(p0, seg, t);
      const d = Math.hypot(p.x-point.x, p.y-point.y);
      if(!best || d<best.dist) best = {dist:d, segIndex:i, t};
    }
    anchor = {x:seg.x, y:seg.y};
  }
  return best;
}
function anchorBefore(pts, segIndex){
  for(let i=segIndex-1;i>=0;i--){
    if(["M","L","C","Q"].includes(pts[i].type)) return {x:pts[i].x, y:pts[i].y};
  }
  return null;
}
function splitSegment(pts, segIndex, t){
  const seg = pts[segIndex];
  const p0 = anchorBefore(pts, segIndex);
  if(!p0 || seg.type==="Z") return pts;
  const lerp = (a,b,t)=>({x:a.x+(b.x-a.x)*t, y:a.y+(b.y-a.y)*t});
  let newSeg1, newSeg2;
  if(seg.type==="L"){
    const mid = lerp(p0, seg, t);
    newSeg1 = {type:"L", x:mid.x, y:mid.y};
    newSeg2 = {type:"L", x:seg.x, y:seg.y};
  } else if(seg.type==="C"){
    const C1={x:seg.c1x,y:seg.c1y}, C2={x:seg.c2x,y:seg.c2y}, P3={x:seg.x,y:seg.y};
    const Q1=lerp(p0,C1,t), H=lerp(C1,C2,t), Q2=lerp(C2,P3,t);
    const R1=lerp(Q1,H,t), R2=lerp(H,Q2,t), S=lerp(R1,R2,t);
    newSeg1 = {type:"C", c1x:Q1.x,c1y:Q1.y, c2x:R1.x,c2y:R1.y, x:S.x, y:S.y};
    newSeg2 = {type:"C", c1x:R2.x,c1y:R2.y, c2x:Q2.x,c2y:Q2.y, x:P3.x, y:P3.y};
  } else if(seg.type==="Q"){
    const C={x:seg.c1x,y:seg.c1y}, P1={x:seg.x,y:seg.y};
    const Q1=lerp(p0,C,t), R=lerp(C,P1,t), S=lerp(Q1,R,t);
    newSeg1 = {type:"Q", c1x:Q1.x,c1y:Q1.y, x:S.x, y:S.y};
    newSeg2 = {type:"Q", c1x:R.x,c1y:R.y, x:P1.x, y:P1.y};
  } else return pts;
  const out = pts.slice();
  out.splice(segIndex, 1, newSeg1, newSeg2);
  return out;
}
function bendLineToCurve(pts, segIndex, dragPoint){
  const p0 = anchorBefore(pts, segIndex);
  const seg = pts[segIndex];
  if(!p0 || seg.type!=="L") return pts;
  const P3 = {x:seg.x, y:seg.y};
  const Q = {x: 2*dragPoint.x - 0.5*(p0.x+P3.x), y: 2*dragPoint.y - 0.5*(p0.y+P3.y)};
  const C1 = {x: p0.x + (Q.x-p0.x)*2/3, y: p0.y + (Q.y-p0.y)*2/3};
  const C2 = {x: P3.x + (Q.x-P3.x)*2/3, y: P3.y + (Q.y-P3.y)*2/3};
  const out = pts.slice();
  out[segIndex] = {type:"C", c1x:C1.x, c1y:C1.y, c2x:C2.x, c2y:C2.y, x:P3.x, y:P3.y};
  return out;
}
function enterNodeEdit(el){
  if(el.tagName!=="path"){ return; }
  nodeEditTarget = el;
  renderOverlay();
}
function exitNodeEdit(){
  nodeEditTarget = null;
  renderOverlay();
}
function renderNodeHandles(){
  const pts = pathToPoints(nodeEditTarget);
  pts.forEach((p,i)=>{
    if(p.type==="Z") return;
    if(p.c1x!==undefined){
      const s1 = svgToScreen(p.c1x,p.c1y);
      const anchorPrev = pts[i-1];
      if(anchorPrev){
        const sA = svgToScreen(anchorPrev.x, anchorPrev.y);
        const line = document.createElementNS(SVGNS,"line");
        line.setAttribute("x1",sA.x); line.setAttribute("y1",sA.y);
        line.setAttribute("x2",s1.x); line.setAttribute("y2",s1.y);
        line.setAttribute("class","node-line");
        selOverlay.appendChild(line);
      }
      const c1 = document.createElementNS(SVGNS,"circle");
      c1.setAttribute("cx",s1.x); c1.setAttribute("cy",s1.y);
      c1.setAttribute("class","node-ctrl");
      c1.dataset.idx=i; c1.dataset.field="c1";
      selOverlay.appendChild(c1);
    }
    if(p.c2x!==undefined){
      const s2 = svgToScreen(p.c2x,p.c2y);
      const sE = svgToScreen(p.x,p.y);
      const line = document.createElementNS(SVGNS,"line");
      line.setAttribute("x1",sE.x); line.setAttribute("y1",sE.y);
      line.setAttribute("x2",s2.x); line.setAttribute("y2",s2.y);
      line.setAttribute("class","node-line");
      selOverlay.appendChild(line);
      const c2 = document.createElementNS(SVGNS,"circle");
      c2.setAttribute("cx",s2.x); c2.setAttribute("cy",s2.y);
      c2.setAttribute("class","node-ctrl");
      c2.dataset.idx=i; c2.dataset.field="c2";
      selOverlay.appendChild(c2);
    }
    const sp = svgToScreen(p.x,p.y);
    const dot = document.createElementNS(SVGNS,"rect");
    dot.setAttribute("x",sp.x-4); dot.setAttribute("y",sp.y-4);
    dot.setAttribute("width",8); dot.setAttribute("height",8);
    dot.setAttribute("class","node-pt");
    dot.dataset.idx=i; dot.dataset.field="anchor";
    selOverlay.appendChild(dot);
  });
  attachNodeEvents();
}
function attachNodeEvents(){
  selOverlay.querySelectorAll(".node-pt,.node-ctrl").forEach(h=>{
    h.addEventListener("pointerdown", ev=>{
      ev.stopPropagation();
      const idx = +h.dataset.idx, field = h.dataset.field;
      const pts = pathToPoints(nodeEditTarget);
      const onMove = mv=>{
        const p = screenToSvg(mv.clientX, mv.clientY);
        if(field==="anchor"){ pts[idx].x=p.x; pts[idx].y=p.y; }
        else if(field==="c1"){ pts[idx].c1x=p.x; pts[idx].c1y=p.y; }
        else if(field==="c2"){ pts[idx].c2x=p.x; pts[idx].c2y=p.y; }
        nodeEditTarget.setAttribute("d", pointsToPath(pts));
        renderOverlay();
      };
      const onUp = ()=>{
        window.removeEventListener("pointermove",onMove);
        window.removeEventListener("pointerup",onUp);
        pushHistory();
      };
      window.addEventListener("pointermove",onMove);
      window.addEventListener("pointerup",onUp);
    });
    h.addEventListener("dblclick", ev=>{
      ev.stopPropagation();
      if(h.dataset.field!=="anchor") return;
      const idx = +h.dataset.idx;
      const pts = pathToPoints(nodeEditTarget);
      if(pts[idx].type!=="Z"){ pts.splice(idx,1); }
      nodeEditTarget.setAttribute("d", pointsToPath(pts));
      pushHistory();
      renderOverlay();
    });
  });
}

// ---------- Handle drag (resize/rotate/move) ----------
function attachHandleEvents(){
  selOverlay.querySelectorAll(".handle,.rot-handle").forEach(h=>{
    h.addEventListener("pointerdown", ev=>{
      ev.stopPropagation();
      const handleId = h.dataset.handle;
      const el = selected[0];
      const bb = getBBox(el);
      const startPt = screenToSvg(ev.clientX, ev.clientY);
      const start = {x:bb.x,y:bb.y,w:bb.width,h:bb.height};
      const cx = bb.x+bb.width/2, cy = bb.y+bb.height/2;

      const onMove = mv=>{
        const p = screenToSvg(mv.clientX, mv.clientY);
        if(handleId==="rot"){
          const ang = Math.atan2(p.y-cy, p.x-cx)*180/Math.PI + 90;
          el.setAttribute("transform", `rotate(${ang.toFixed(1)} ${cx} ${cy})`);
        } else {
          resizeElement(el, start, +handleId, p, startPt);
        }
        renderOverlay();
        updateTransformPanel();
      };
      const onUp = ()=>{
        window.removeEventListener("pointermove",onMove);
        window.removeEventListener("pointerup",onUp);
        pushHistory();
      };
      window.addEventListener("pointermove",onMove);
      window.addEventListener("pointerup",onUp);
    });
  });
}
function resizeElement(el, start, handleId, p){
  // handles: 0=TL 1=TR 2=BL 3=BR 4=TM 5=BM 6=ML 7=MR
  let nx=start.x, ny=start.y, nw=start.w, nh=start.h;
  const right = start.x+start.w, bottom = start.y+start.h;
  if([0,4,6].includes(handleId)===false){} // no-op placeholder
  if([0,2,6].includes(handleId)){ nx = p.x; nw = right - p.x; }
  if([1,3,7].includes(handleId)){ nw = p.x - start.x; }
  if([0,1,4].includes(handleId)){ ny = p.y; nh = bottom - p.y; }
  if([2,3,5].includes(handleId)){ nh = p.y - start.y; }
  if(nw<2) nw=2; if(nh<2) nh=2;
  applyBBoxToElement(el, nx, ny, nw, nh);
}
function applyBBoxToElement(el, x, y, w, h){
  const tag = el.tagName;
  if(tag==="rect"){
    el.setAttribute("x",x); el.setAttribute("y",y);
    el.setAttribute("width",w); el.setAttribute("height",h);
  } else if(tag==="ellipse"){
    el.setAttribute("cx", x+w/2); el.setAttribute("cy", y+h/2);
    el.setAttribute("rx", w/2); el.setAttribute("ry", h/2);
  } else if(tag==="line"){
    // scale endpoints proportionally within original bbox
    const bb = getBBox(el);
    const sx = bb.width ? w/bb.width : 1, sy = bb.height ? h/bb.height : 1;
    const x1=+el.getAttribute("x1"), y1=+el.getAttribute("y1"), x2=+el.getAttribute("x2"), y2=+el.getAttribute("y2");
    el.setAttribute("x1", x+(x1-bb.x)*sx); el.setAttribute("y1", y+(y1-bb.y)*sy);
    el.setAttribute("x2", x+(x2-bb.x)*sx); el.setAttribute("y2", y+(y2-bb.y)*sy);
  } else if(tag==="path"){
    const bb = getBBox(el);
    const sx = bb.width ? w/bb.width : 1, sy = bb.height ? h/bb.height : 1;
    const pts = pathToPoints(el);
    pts.forEach(pt=>{
      if(pt.x!==undefined){ pt.x = x+(pt.x-bb.x)*sx; pt.y = y+(pt.y-bb.y)*sy; }
      if(pt.c1x!==undefined){ pt.c1x = x+(pt.c1x-bb.x)*sx; pt.c1y = y+(pt.c1y-bb.y)*sy; }
      if(pt.c2x!==undefined){ pt.c2x = x+(pt.c2x-bb.x)*sx; pt.c2y = y+(pt.c2y-bb.y)*sy; }
    });
    el.setAttribute("d", pointsToPath(pts));
  } else if(tag==="text"){
    el.setAttribute("x", x); el.setAttribute("y", y+h);
  }
}

// ---------- Marquee select ----------
let marqueeEl = null, marqueeStart = null;
function startMarquee(p){
  marqueeStart = p;
  marqueeEl = document.createElementNS(SVGNS,"rect");
  marqueeEl.setAttribute("class","sel-box");
  marqueeEl.setAttribute("stroke","#4f8cff");
  selOverlay.appendChild(marqueeEl);
}
function updateMarquee(p){
  const s1 = svgToScreen(Math.min(marqueeStart.x,p.x), Math.min(marqueeStart.y,p.y));
  const s2 = svgToScreen(Math.max(marqueeStart.x,p.x), Math.max(marqueeStart.y,p.y));
  marqueeEl.setAttribute("x",s1.x); marqueeEl.setAttribute("y",s1.y);
  marqueeEl.setAttribute("width",s2.x-s1.x); marqueeEl.setAttribute("height",s2.y-s1.y);
  return {x1:Math.min(marqueeStart.x,p.x), y1:Math.min(marqueeStart.y,p.y), x2:Math.max(marqueeStart.x,p.x), y2:Math.max(marqueeStart.y,p.y)};
}
function finishMarquee(box){
  const hits = [];
  Array.from(drawLayer.children).forEach(el=>{
    const bb = getBBox(el);
    if(bb.x>=box.x1 && bb.y>=box.y1 && bb.x+bb.width<=box.x2 && bb.y+bb.height<=box.y2){
      hits.push(el);
    }
  });
  setSelection(hits);
  marqueeEl && marqueeEl.remove();
  marqueeEl = null;
}

const SHAPE_API_ROOT = "https://api.github.com/repos/lukeo25/WickTools/contents/scripts/Panel_SVG/Shapes";
let shapePath = "";
const shapeDrawer = document.getElementById("shapeDrawer");
const shapeGrid = document.getElementById("shapeGrid");

const shapeTitle = document.getElementById("shapeDrawerTitle");

function openShapesDrawer(){

}

document.getElementById("shapeClose").addEventListener("click", closeShapesDrawer);
document.getElementById("shapeBack").addEventListener("click", ()=>{
  if(!shapePath){ closeShapesDrawer(); return; }
  const bits = shapePath.split("/").filter(Boolean);
  bits.pop();
  loadShapeFolder(bits.join("/"));
});

async function loadShapeFolder(path){
  shapePath = path || "";
  shapeGrid.innerHTML = "";
  shapeTitle.textContent = shapePath ? "Shapes · " + shapePath.split("/").pop() : "LukeTools Shapes";
  shapeStatus.textContent = "Loading…";
  const url = SHAPE_API_ROOT + (shapePath ? "/" + shapePath.split("/").map(encodeURIComponent).join("/") : "") + "?ref=main";
  try{
    const r = await fetch(url,{cache:"no-store"});
    if(!r.ok) throw new Error("GitHub HTTP " + r.status);
    const items = await r.json();
    const dirs = items.filter(x=>x.type==="dir").sort((a,b)=>a.name.localeCompare(b.name));
    const svgs = items.filter(x=>x.type==="file" && /\.svg$/i.test(x.name)).sort((a,b)=>a.name.localeCompare(b.name));
    dirs.forEach(item=>{
      const b=document.createElement("button");
      b.className="shape-folder";
      b.innerHTML="<span>▣</span><span>"+item.name.replace(/[_-]+/g," ")+"</span>";
      b.addEventListener("click",()=>loadShapeFolder(shapePath ? shapePath+"/"+item.name : item.name));
      shapeGrid.appendChild(b);
    });
    svgs.forEach(item=>{
      const b=document.createElement("button");
      b.className="shape-card";
      b.title=item.name.replace(/\.svg$/i,"").replace(/[_-]+/g," ");
      const img=document.createElement("img");
      img.src=item.download_url;
      img.alt=b.title;
      b.appendChild(img);
      b.addEventListener("click",()=>insertLibraryShape(item.download_url,b.title));
      shapeGrid.appendChild(b);
    });
    shapeStatus.textContent = (dirs.length+svgs.length) + " items";
  }catch(err){
    shapeStatus.textContent = "Could not load shapes: " + (err?.message || err);
  }
}

async function insertLibraryShape(url,name){
  try{
    shapeStatus.textContent="Adding "+name+"…";
    const r=await fetch(url,{cache:"no-store"});
    if(!r.ok) throw new Error("HTTP "+r.status);
    const txt=await r.text();
    const doc=new DOMParser().parseFromString(txt,"image/svg+xml");
    const root=doc.documentElement;
    if(!root || root.tagName.toLowerCase()!=="svg") throw new Error("Invalid SVG");
    const vb=(root.getAttribute("viewBox")||"0 0 100 100").trim().split(/[\s,]+/).map(Number);
    const w=(vb[2]||100), h=(vb[3]||100);
    const target=150;
    const scale=Math.min(target/w,target/h);
    const cr=canvasWrap.getBoundingClientRect();
    const center=screenToSvg(cr.left+cr.width/2,cr.top+cr.height/2);
    const g=document.createElementNS(SVGNS,"g");
    g.id=newId();
    g.dataset.libraryShape=name;
    g.setAttribute("transform",`translate(${center.x-(w*scale)/2} ${center.y-(h*scale)/2}) scale(${scale}) translate(${-(vb[0]||0)} ${-(vb[1]||0)})`);
    Array.from(root.childNodes).forEach(n=>{
      if(n.nodeType===1) g.appendChild(document.importNode(n,true));
    });
    drawLayer.appendChild(g);
    setSelection([g]);
    pushHistory();
    shapeStatus.textContent="Added "+name;
  }catch(err){
    shapeStatus.textContent="Shape failed: "+(err?.message||err);
  }
}

function texDist(a,b){return Math.hypot(b.x-a.x,b.y-a.y)}
function texNorm(x,y){const m=Math.hypot(x,y)||1;return{x:x/m,y:y/m}}
function texRng(seed){let s=(seed|0)||1234567;return()=>{s=(s*1664525+1013904223)|0;return((s>>>0)/4294967296)}}
function texTaperFactor(i,n,taper){
  if(n<2) return 1;
  const u=i/(n-1);
  const edge=Math.min(u,1-u)*2;
  return (1-taper)+taper*Math.max(.04,Math.sin(Math.PI*Math.min(1,edge)));
}
function buildTextureOutline(points,settings){
  if(!points || points.length<2) return null;
  const sm=[points[0]];
  for(let i=1;i<points.length-1;i++){
    const a=points[i-1],b=points[i],c=points[i+1];
    sm.push({x:b.x+( (a.x+b.x+c.x)/3-b.x)*settings.smoothing,
             y:b.y+( (a.y+b.y+c.y)/3-b.y)*settings.smoothing,
             pressure:b.pressure||.5});
  }
  sm.push(points[points.length-1]);
  const rng=texRng(sm.length*911+Math.round(settings.roughness*1000)*17);
  const left=[],right=[];
  for(let i=0;i<sm.length;i++){
    const prv=sm[Math.max(0,i-1)],nxt=sm[Math.min(sm.length-1,i+1)];
    const t=texNorm(nxt.x-prv.x,nxt.y-prv.y), n={x:-t.y,y:t.x};
    const pressure=sm[i].pressure||.5;
    let width=settings.size*(.80+.35*pressure)*texTaperFactor(i,sm.length,settings.taper)*.5;
    const edgeNoise=settings.roughness*settings.size*(rng()-.5)*.9;
    const center=settings.jitter*settings.size*(rng()-.5)*.55;
    const cx=sm[i].x+n.x*center,cy=sm[i].y+n.y*center;
    const off=Math.max(.6,width+edgeNoise);
    left.push({x:cx+n.x*off,y:cy+n.y*off});
    right.push({x:cx-n.x*off,y:cy-n.y*off});
  }
  return left.concat(right.reverse());
}
function texturePolyToD(poly){
  if(!poly || !poly.length) return "";
  return "M "+poly.map(p=>p.x.toFixed(2)+" "+p.y.toFixed(2)).join(" L ")+" Z";
}

  ["fill","stroke","stroke-width","opacity","transform"].forEach(attr=>{

    if(v!==null) path.setAttribute(attr, v); else path.removeAttribute(attr);

  el.parentNode.insertBefore(path, el);
  el.remove();
  return path;
}

  }
  if(m){

  }
}

function normalizeToCubic(pts){

  for(let i=0;i<pts.length;i++){
    const seg = pts[i];

    if(seg.type==="Z"){ out.push(Object.assign({},seg)); continue; }
    if(seg.type==="Q"){
      const P0=anchor, C={x:seg.c1x,y:seg.c1y}, P1={x:seg.x,y:seg.y};
      const c1={x:P0.x+(C.x-P0.x)*2/3, y:P0.y+(C.y-P0.y)*2/3};
      const c2={x:P1.x+(C.x-P1.x)*2/3, y:P1.y+(C.y-P1.y)*2/3};
      out.push({type:"C", c1x:c1.x,c1y:c1.y,c2x:c2.x,c2y:c2.y,x:P1.x,y:P1.y});
      anchor = P1;
      continue;
    }
    out.push(Object.assign({},seg));
    anchor = {x:seg.x, y:seg.y};
  }
  return out;
}
function ptsToPaperSegments(pts){
  const norm = normalizeToCubic(pts);
  const anchors = [];
  let prevAnchor = null;
  for(let i=0;i<norm.length;i++){
    const seg = norm[i];
    if(seg.type==="M"){ const a={x:seg.x,y:seg.y,hIn:null,hOut:null}; anchors.push(a); prevAnchor=a; continue; }
    if(seg.type==="Z"){ continue; }
    if(seg.type==="L"){ const a={x:seg.x,y:seg.y,hIn:null,hOut:null}; anchors.push(a); prevAnchor=a; continue; }
    if(seg.type==="C"){
      if(prevAnchor) prevAnchor.hOut = {x:seg.c1x,y:seg.c1y};
      const a = {x:seg.x, y:seg.y, hIn:{x:seg.c2x,y:seg.c2y}, hOut:null};
      anchors.push(a);
      prevAnchor = a;
    }
  }
  const closed = pts.some(p=>p.type==="Z");
  const segments = anchors.map(a=>{
    const hi = a.hIn ? [a.hIn.x-a.x, a.hIn.y-a.y] : [0,0];
    const ho = a.hOut ? [a.hOut.x-a.x, a.hOut.y-a.y] : [0,0];
    if(hi[0]===0 && hi[1]===0 && ho[0]===0 && ho[1]===0) return [a.x,a.y];
    return [a.x,a.y, hi[0],hi[1], ho[0],ho[1]];
  });
  return {segments, closed};
}
function hexToRgbArr(hex){
  if(!hex || hex==="none") return null;
  const h = hex.replace("#","");
  if(h.length<6) return null;
  return [parseInt(h.substring(0,2),16)/255, parseInt(h.substring(2,4),16)/255, parseInt(h.substring(4,6),16)/255, 1];
}
function elToPathPoints(el){
  if(el.tagName==="path") return pathToPoints(el);
  if(el.tagName==="rect"){
    const x=+el.getAttribute("x"), y=+el.getAttribute("y"), w=+el.getAttribute("width"), h=+el.getAttribute("height");
    return [{type:"M",x,y},{type:"L",x:x+w,y},{type:"L",x:x+w,y:y+h},{type:"L",x,y:y+h},{type:"Z"}];
  }
  if(el.tagName==="ellipse"){
    const cx=+el.getAttribute("cx"), cy=+el.getAttribute("cy"), rx=+el.getAttribute("rx"), ry=+el.getAttribute("ry");
    const k=0.5522847498;
    return [
      {type:"M", x:cx+rx, y:cy},
      {type:"C", c1x:cx+rx, c1y:cy+ry*k, c2x:cx+rx*k, c2y:cy+ry, x:cx, y:cy+ry},
      {type:"C", c1x:cx-rx*k, c1y:cy+ry, c2x:cx-rx, c2y:cy+ry*k, x:cx-rx, y:cy},
      {type:"C", c1x:cx-rx, c1y:cy-ry*k, c2x:cx-rx*k, c2y:cy-ry, x:cx, y:cy-ry},
      {type:"C", c1x:cx+rx*k, c1y:cy-ry, c2x:cx+rx, c2y:cy-ry*k, x:cx+rx, y:cy},
      {type:"Z"}
    ];
  }
  if(el.tagName==="line"){
    return [{type:"M", x:+el.getAttribute("x1"), y:+el.getAttribute("y1")}, {type:"L", x:+el.getAttribute("x2"), y:+el.getAttribute("y2")}];
  }
  return null;
}
function exportSelectedAsPaperJSON(){
  if(!selected.length){ alert("Select one or more shapes first."); return; }
  const results = selected.map(el=>{
    const pts = elToPathPoints(el);
    if(!pts) return null;
    const {segments, closed} = ptsToPaperSegments(pts);
    const props = {applyMatrix:true, segments, closed};
    const fc = hexToRgbArr(el.getAttribute("fill")); if(fc) props.fillColor = fc;
    const sc = hexToRgbArr(el.getAttribute("stroke")); if(sc) props.strokeColor = sc;
    const sw = el.getAttribute("stroke-width"); if(sw && sc) props.strokeWidth = +sw;
    return ["Path", props];
  }).filter(Boolean);
  if(!results.length){ alert("Nothing exportable in the current selection."); return; }
  const json = results.length===1 ? JSON.stringify(results[0]) : JSON.stringify(["Group", {applyMatrix:true, children:results}]);
  const done = ()=> flashStatus("Copied Paper.js curve JSON — in Wick, run: project.activeLayer.importJSON(json)");
  if(navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(json).then(done).catch(()=>prompt("Copy this Paper.js/Wick JSON:", json));
  } else {
    prompt("Copy this Paper.js/Wick JSON:", json);
  }
}

    if(loadedSvg.tagName!=="svg"){ alert("Not a valid SVG file."); return; }

    else { pageW = parseFloat(loadedSvg.getAttribute("width"))||1280; pageH = parseFloat(loadedSvg.getAttribute("height"))||800; }

    document.getElementById("pageH").value = pageH;
    pageRect.setAttribute("width", pageW); pageRect.setAttribute("height", pageH);
    Array.from(loadedSvg.children).forEach(child=>{
      if(child.tagName==="rect" && child.id==="pageRect") return;
      const imported = document.importNode(child, true);
      if(!imported.id) imported.id = newId();
      drawLayer.appendChild(imported);
    });
    clearSelection();
    fitToScreen();
    history=[]; historyIndex=-1;
    pushHistory();
  };
  reader.readAsText(file);
  ev.target.value = "";
});

function downloadBlob(blob, filename){
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url; a.download = filename;
  document.body.appendChild(a); a.click(); a.remove();
  URL.revokeObjectURL(url);
}
