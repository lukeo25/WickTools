// ---------- Last session persistence ----------
const DRAWING_SESSION_DB = "LukeToolsDrawingSession";
const DRAWING_SESSION_STORE = "session";
const DRAWING_SESSION_KEY = "last";
let drawingSessionDbPromise = null;
let drawingSessionSaveTimer = null;

function openDrawingSessionDb(){
  if(drawingSessionDbPromise) return drawingSessionDbPromise;
  drawingSessionDbPromise = new Promise((resolve,reject)=>{
    const req = indexedDB.open(DRAWING_SESSION_DB,1);
    req.onupgradeneeded = ()=>{
      const db=req.result;
      if(!db.objectStoreNames.contains(DRAWING_SESSION_STORE)){
        db.createObjectStore(DRAWING_SESSION_STORE,{keyPath:"id"});
      }
    };
    req.onsuccess=()=>resolve(req.result);
    req.onerror=()=>reject(req.error || new Error("Could not open Drawing session storage"));
  });
  return drawingSessionDbPromise;
}

async function drawingSessionPut(record){
  const db=await openDrawingSessionDb();
  return new Promise((resolve,reject)=>{
    const tx=db.transaction(DRAWING_SESSION_STORE,"readwrite");
    tx.objectStore(DRAWING_SESSION_STORE).put(record);
    tx.oncomplete=()=>resolve(record);
    tx.onerror=()=>reject(tx.error);
  });
}

async function drawingSessionGet(){
  const db=await openDrawingSessionDb();
  return new Promise((resolve,reject)=>{
    const tx=db.transaction(DRAWING_SESSION_STORE,"readonly");
    const req=tx.objectStore(DRAWING_SESSION_STORE).get(DRAWING_SESSION_KEY);
    req.onsuccess=()=>resolve(req.result||null);
    req.onerror=()=>reject(req.error);
  });
}

function scheduleDrawingSessionSave(){
  clearTimeout(drawingSessionSaveTimer);
  drawingSessionSaveTimer=setTimeout(()=>{
    drawingSessionPut({
      id:DRAWING_SESSION_KEY,
      savedAt:Date.now(),
      pageW:Number(pageW)||1280,
      pageH:Number(pageH)||800,
      drawing:drawLayer.innerHTML,
      drawingFrames:(()=>{
        try{ captureCurrentDrawingFrame(); }catch(e){}
        return drawingFrames.map(frame=>({id:frame.id,label:frame.label,drawing:frame.drawing}));
      })(),
      drawingFrameIndex:Number(drawingFrameIndex)||0,
      drawingFrameNextId:Number(drawingFrameNextId)||2,
      drawingGizmoMode:!!drawingGizmoMode,
      drawingOnionSkin:!!drawingOnionSkin,
      zoom:Number(zoom)||1,
      panX:Number(panX)||0,
      panY:Number(panY)||0
    }).catch(err=>console.warn("[Drawing] Could not save last session",err));
  },120);
}

async function restoreDrawingSession(){
  try{
    const saved=await drawingSessionGet();
    if(!saved || typeof saved.drawing!=="string") return false;

    pageW=Math.max(10,Number(saved.pageW)||1280);
    pageH=Math.max(10,Number(saved.pageH)||800);
    document.getElementById("pageW").value=pageW;
    document.getElementById("pageH").value=pageH;
    pageRect.setAttribute("width",pageW);
    pageRect.setAttribute("height",pageH);
    svgRoot.setAttribute("width",pageW);
    svgRoot.setAttribute("height",pageH);
    svgRoot.setAttribute("viewBox",`0 0 ${pageW} ${pageH}`);
    stage.style.width=pageW+"px";
    stage.style.height=pageH+"px";
    if(Array.isArray(saved.drawingFrames) && saved.drawingFrames.length){
      drawingFrames=saved.drawingFrames.map((frame,index)=>({
        id:Number(frame&&frame.id)||index+1,
        label:String(frame&&frame.label||("Frame "+(index+1))),
        drawing:String(frame&&frame.drawing||"")
      }));
      drawingFrameIndex=Math.max(0,Math.min(drawingFrames.length-1,Number(saved.drawingFrameIndex)||0));
      drawingFrameNextId=Math.max(Number(saved.drawingFrameNextId)||2,drawingFrames.reduce((m,f)=>Math.max(m,Number(f.id)||0),0)+1);
      drawLayer.innerHTML=drawingFrames[drawingFrameIndex].drawing;
    }else{
      drawingFrames=[{id:1,label:"Frame 1",drawing:String(saved.drawing||"")}];
      drawingFrameIndex=0;
      drawingFrameNextId=2;
      drawLayer.innerHTML=saved.drawing;
    }
    drawingGizmoMode=!!saved.drawingGizmoMode;
    drawingOnionSkin=saved.drawingOnionSkin!==false;

    zoom=isFinite(Number(saved.zoom)) ? Math.max(.05,Math.min(16,Number(saved.zoom))) : 1;
    panX=isFinite(Number(saved.panX)) ? Number(saved.panX) : 60;
    panY=isFinite(Number(saved.panY)) ? Number(saved.panY) : 40;
    applyStageTransform();

    clearSelection();
    history=[];
    historyIndex=-1;
    history.push(snapshot());
    historyIndex=0;
    updateLayerList();
    updateStatusCount();
    renderDrawingFrames();
    renderDrawingOnionSkin();
    return true;
  }catch(err){
    console.warn("[Drawing] Could not restore last session",err);
    return false;
  }
}

function pushHistory(){
  history = history.slice(0, historyIndex+1);
  history.push(snapshot());
  if(history.length>100) history.shift();
  historyIndex = history.length-1;
  updateLayerList();
  updateStatusCount();
  scheduleDrawingSessionSave();
}
function undo(){
  if(historyIndex<=0) return;
  historyIndex--;
  drawLayer.innerHTML = history[historyIndex];
  clearSelection();
  updateLayerList();
  updateStatusCount();
  scheduleDrawingSessionSave();
}
function redo(){
  if(historyIndex>=history.length-1) return;
  historyIndex++;
  drawLayer.innerHTML = history[historyIndex];
  clearSelection();
  updateLayerList();
  updateStatusCount();
  scheduleDrawingSessionSave();
}

  try { return el.getBBox(); } catch(e){ return {x:0,y:0,width:0,height:0}; }

function getVisualBBox(el){
  try{
    const bb=el.getBBox();
    const m=el.getCTM();
    if(!m) return bb;
    const pts=[
      new DOMPoint(bb.x,bb.y).matrixTransform(m),
      new DOMPoint(bb.x+bb.width,bb.y).matrixTransform(m),
      new DOMPoint(bb.x+bb.width,bb.y+bb.height).matrixTransform(m),
      new DOMPoint(bb.x,bb.y+bb.height).matrixTransform(m)
    ];
    const xs=pts.map(p=>p.x), ys=pts.map(p=>p.y);
    const x=Math.min(...xs), y=Math.min(...ys);
    const x2=Math.max(...xs), y2=Math.max(...ys);
    return {x,y,width:x2-x,height:y2-y};
  }catch(e){
    return getBBox(el);
  }
}

// ---------- Node editing (paths) ----------
function pathToPoints(el){
  // Parses simple M/L/C/Q/Z path data into an editable point list.
  const d = el.getAttribute("d") || "";
  const cmds = d.match(/[MLCQZ][^MLCQZ]*/gi) || [];
  const pts = [];
  let cur = {x:0,y:0};
  cmds.forEach(c=>{
    const type = c[0];
    const nums = c.slice(1).trim().split(/[\s,]+/).filter(s=>s.length).map(Number);
    if(type.toUpperCase()==="M"){ cur={x:nums[0],y:nums[1]}; pts.push({type:"M", x:nums[0], y:nums[1]}); }
    else if(type.toUpperCase()==="L"){ cur={x:nums[0],y:nums[1]}; pts.push({type:"L", x:nums[0], y:nums[1]}); }
    else if(type.toUpperCase()==="C"){ pts.push({type:"C", c1x:nums[0],c1y:nums[1],c2x:nums[2],c2y:nums[3],x:nums[4],y:nums[5]}); cur={x:nums[4],y:nums[5]}; }
    else if(type.toUpperCase()==="Q"){ pts.push({type:"Q", c1x:nums[0],c1y:nums[1],x:nums[2],y:nums[3]}); cur={x:nums[2],y:nums[3]}; }
    else if(type.toUpperCase()==="Z"){ pts.push({type:"Z"}); }
  });
  return pts;
}
function pointsToPath(pts){
  return pts.map(p=>{
    if(p.type==="M") return `M ${p.x} ${p.y}`;
    if(p.type==="L") return `L ${p.x} ${p.y}`;
    if(p.type==="C") return `C ${p.c1x} ${p.c1y} ${p.c2x} ${p.c2y} ${p.x} ${p.y}`;
    if(p.type==="Q") return `Q ${p.c1x} ${p.c1y} ${p.x} ${p.y}`;
    if(p.type==="Z") return "Z";
    return "";
  }).join(" ");
}
function evalSegment(p0, seg, t){
  const mt = 1-t;
  if(seg.type==="L") return {x:p0.x+(seg.x-p0.x)*t, y:p0.y+(seg.y-p0.y)*t};
  if(seg.type==="C") return {
    x: mt*mt*mt*p0.x + 3*mt*mt*t*seg.c1x + 3*mt*t*t*seg.c2x + t*t*t*seg.x,
    y: mt*mt*mt*p0.y + 3*mt*mt*t*seg.c1y + 3*mt*t*t*seg.c2y + t*t*t*seg.y
  };
  if(seg.type==="Q") return {
    x: mt*mt*p0.x + 2*mt*t*seg.c1x + t*t*seg.x,
    y: mt*mt*p0.y + 2*mt*t*seg.c1y + t*t*seg.y
  };
  return p0;
}
function findClosestSegment(pts, point){
  let best = null, anchor = null, subpathStart = null;
  for(let i=0;i<pts.length;i++){
    const seg = pts[i];
    if(seg.type==="M"){ anchor = {x:seg.x,y:seg.y}; subpathStart = anchor; continue; }
    if(seg.type==="Z"){ anchor = subpathStart; continue; }
    const p0 = anchor;
    for(let s=0;s<=16;s++){
      const t = s/16;
      const p = evalSegment(p0, seg, t);
      const d = Math.hypot(p.x-point.x, p.y-point.y);
      if(!best || d<best.dist) best = {dist:d, segIndex:i, t};
    }
    anchor = {x:seg.x, y:seg.y};
  }
  return best;
}
function anchorBefore(pts, segIndex){
  for(let i=segIndex-1;i>=0;i--){
    if(["M","L","C","Q"].includes(pts[i].type)) return {x:pts[i].x, y:pts[i].y};
  }
  return null;
}
function splitSegment(pts, segIndex, t){
  const seg = pts[segIndex];
  const p0 = anchorBefore(pts, segIndex);
  if(!p0 || seg.type==="Z") return pts;
  const lerp = (a,b,t)=>({x:a.x+(b.x-a.x)*t, y:a.y+(b.y-a.y)*t});
  let newSeg1, newSeg2;
  if(seg.type==="L"){
    const mid = lerp(p0, seg, t);
    newSeg1 = {type:"L", x:mid.x, y:mid.y};
    newSeg2 = {type:"L", x:seg.x, y:seg.y};
  } else if(seg.type==="C"){
    const C1={x:seg.c1x,y:seg.c1y}, C2={x:seg.c2x,y:seg.c2y}, P3={x:seg.x,y:seg.y};
    const Q1=lerp(p0,C1,t), H=lerp(C1,C2,t), Q2=lerp(C2,P3,t);
    const R1=lerp(Q1,H,t), R2=lerp(H,Q2,t), S=lerp(R1,R2,t);
    newSeg1 = {type:"C", c1x:Q1.x,c1y:Q1.y, c2x:R1.x,c2y:R1.y, x:S.x, y:S.y};
    newSeg2 = {type:"C", c1x:R2.x,c1y:R2.y, c2x:Q2.x,c2y:Q2.y, x:P3.x, y:P3.y};
  } else if(seg.type==="Q"){
    const C={x:seg.c1x,y:seg.c1y}, P1={x:seg.x,y:seg.y};
    const Q1=lerp(p0,C,t), R=lerp(C,P1,t), S=lerp(Q1,R,t);
    newSeg1 = {type:"Q", c1x:Q1.x,c1y:Q1.y, x:S.x, y:S.y};
    newSeg2 = {type:"Q", c1x:R.x,c1y:R.y, x:P1.x, y:P1.y};
  } else return pts;
  const out = pts.slice();
  out.splice(segIndex, 1, newSeg1, newSeg2);
  return out;
}
function bendLineToCurve(pts, segIndex, dragPoint){
  const p0 = anchorBefore(pts, segIndex);
  const seg = pts[segIndex];
  if(!p0 || seg.type!=="L") return pts;
  const P3 = {x:seg.x, y:seg.y};
  const Q = {x: 2*dragPoint.x - 0.5*(p0.x+P3.x), y: 2*dragPoint.y - 0.5*(p0.y+P3.y)};
  const C1 = {x: p0.x + (Q.x-p0.x)*2/3, y: p0.y + (Q.y-p0.y)*2/3};
  const C2 = {x: P3.x + (Q.x-P3.x)*2/3, y: P3.y + (Q.y-P3.y)*2/3};
  const out = pts.slice();
  out[segIndex] = {type:"C", c1x:C1.x, c1y:C1.y, c2x:C2.x, c2y:C2.y, x:P3.x, y:P3.y};
  return out;
}

function attachNodeEvents(){
  selOverlay.querySelectorAll(".node-pt,.node-ctrl").forEach(h=>{
    h.addEventListener("pointerdown", ev=>{
      ev.stopPropagation();
      const idx = +h.dataset.idx, field = h.dataset.field;
      const pts = pathToPoints(nodeEditTarget);
      const onMove = mv=>{
        const p = screenToSvg(mv.clientX, mv.clientY);
        if(field==="anchor"){ pts[idx].x=p.x; pts[idx].y=p.y; }
        else if(field==="c1"){ pts[idx].c1x=p.x; pts[idx].c1y=p.y; }
        else if(field==="c2"){ pts[idx].c2x=p.x; pts[idx].c2y=p.y; }
        nodeEditTarget.setAttribute("d", pointsToPath(pts));
        renderOverlay();
      };
      const onUp = ()=>{
        window.removeEventListener("pointermove",onMove);
        window.removeEventListener("pointerup",onUp);
        pushHistory();
      };
      window.addEventListener("pointermove",onMove);
      window.addEventListener("pointerup",onUp);
    });
    h.addEventListener("dblclick", ev=>{
      ev.stopPropagation();
      if(h.dataset.field!=="anchor") return;
      const idx = +h.dataset.idx;
      const pts = pathToPoints(nodeEditTarget);
      if(pts[idx].type!=="Z"){ pts.splice(idx,1); }
      nodeEditTarget.setAttribute("d", pointsToPath(pts));
      pushHistory();
      renderOverlay();
    });
  });
}

// ---------- Handle drag (resize/rotate/move) ----------
function attachHandleEvents(){
  selOverlay.querySelectorAll(".handle,.rot-handle").forEach(h=>{
    h.addEventListener("pointerdown", ev=>{
      ev.stopPropagation();
      const handleId = h.dataset.handle;
      const el = selected[0];
      const bb = getVisualBBox(el);
      const startPt = screenToSvg(ev.clientX, ev.clientY);
      const start = {x:bb.x,y:bb.y,w:bb.width,h:bb.height};
      const cx = bb.x+bb.width/2, cy = bb.y+bb.height/2;
      const startTransform = el.getAttribute("transform") || "";

      const onMove = mv=>{
        const p = screenToSvg(mv.clientX, mv.clientY);
        if(handleId==="rot"){
          const ang = Math.atan2(p.y-cy, p.x-cx)*180/Math.PI + 90;
          if(el.tagName==="g" || startTransform){
            el.setAttribute("transform", `rotate(${ang.toFixed(1)} ${cx} ${cy}) ${startTransform}`.trim());
          } else {
            el.setAttribute("transform", `rotate(${ang.toFixed(1)} ${cx} ${cy})`);
          }
        } else if(el.tagName==="g" || el.tagName==="path"){
          resizeTransformedGroup(el,start,+handleId,p,startTransform);
        } else {
          resizeElement(el, start, +handleId, p, startPt);
        }
        renderOverlay();
        updateTransformPanel();
      };
      const onUp = ()=>{
        window.removeEventListener("pointermove",onMove);
        window.removeEventListener("pointerup",onUp);
        pushHistory();
      };
      window.addEventListener("pointermove",onMove);
      window.addEventListener("pointerup",onUp);
    });
  });
}

function resizeTransformedGroup(el,start,handleId,p,startTransform){
  let nx=start.x, ny=start.y, nw=start.w, nh=start.h;
  const right=start.x+start.w, bottom=start.y+start.h;
  if([0,2,6].includes(handleId)){ nx=p.x; nw=right-p.x; }
  if([1,3,7].includes(handleId)){ nw=p.x-start.x; }
  if([0,1,4].includes(handleId)){ ny=p.y; nh=bottom-p.y; }
  if([2,3,5].includes(handleId)){ nh=p.y-start.y; }

  const minSize=2;
  if(Math.abs(nw)<minSize) nw=(nw<0?-1:1)*minSize;
  if(Math.abs(nh)<minSize) nh=(nh<0?-1:1)*minSize;

  const sx=start.w ? nw/start.w : 1;
  const sy=start.h ? nh/start.h : 1;

  // Map the group's current visual bounding box to the dragged bounding box
  // while preserving its original library transform.
  const tx=nx - start.x*sx;
  const ty=ny - start.y*sy;
  el.setAttribute("transform", `translate(${tx} ${ty}) scale(${sx} ${sy}) ${startTransform}`.trim());
}

function resizeElement(el, start, handleId, p){
  // handles: 0=TL 1=TR 2=BL 3=BR 4=TM 5=BM 6=ML 7=MR
  let nx=start.x, ny=start.y, nw=start.w, nh=start.h;
  const right = start.x+start.w, bottom = start.y+start.h;
  if([0,4,6].includes(handleId)===false){} // no-op placeholder
  if([0,2,6].includes(handleId)){ nx = p.x; nw = right - p.x; }
  if([1,3,7].includes(handleId)){ nw = p.x - start.x; }
  if([0,1,4].includes(handleId)){ ny = p.y; nh = bottom - p.y; }
  if([2,3,5].includes(handleId)){ nh = p.y - start.y; }
  if(nw<2) nw=2; if(nh<2) nh=2;
  applyBBoxToElement(el, nx, ny, nw, nh);
}
function applyBBoxToElement(el, x, y, w, h){
  const tag = el.tagName;
  if(tag==="rect"){
    el.setAttribute("x",x); el.setAttribute("y",y);
    el.setAttribute("width",w); el.setAttribute("height",h);
  } else if(tag==="ellipse"){
    el.setAttribute("cx", x+w/2); el.setAttribute("cy", y+h/2);
    el.setAttribute("rx", w/2); el.setAttribute("ry", h/2);
  } else if(tag==="line"){
    // scale endpoints proportionally within original bbox
    const bb = getBBox(el);
    const sx = bb.width ? w/bb.width : 1, sy = bb.height ? h/bb.height : 1;
    const x1=+el.getAttribute("x1"), y1=+el.getAttribute("y1"), x2=+el.getAttribute("x2"), y2=+el.getAttribute("y2");
    el.setAttribute("x1", x+(x1-bb.x)*sx); el.setAttribute("y1", y+(y1-bb.y)*sy);
    el.setAttribute("x2", x+(x2-bb.x)*sx); el.setAttribute("y2", y+(y2-bb.y)*sy);
  } else if(tag==="path"){
    const bb = getVisualBBox(el);
    const sx = bb.width ? w/bb.width : 1;
    const sy = bb.height ? h/bb.height : 1;
    const base = el.getAttribute("transform") || "";
    const tx = x - bb.x * sx;
    const ty = y - bb.y * sy;
    el.setAttribute("transform", `translate(${tx} ${ty}) scale(${sx} ${sy}) ${base}`.trim());
  } else if(tag==="text"){
    el.setAttribute("x", x); el.setAttribute("y", y+h);
  }
}

let marqueeEl = null, marqueeStart = null;
function startMarquee(p){
  marqueeStart = p;
  marqueeEl = document.createElementNS(SVGNS,"rect");
  marqueeEl.setAttribute("class","sel-box");
  marqueeEl.setAttribute("stroke","#4f8cff");
  selOverlay.appendChild(marqueeEl);
}
function updateMarquee(p){
  const s1 = svgToScreen(Math.min(marqueeStart.x,p.x), Math.min(marqueeStart.y,p.y));
  const s2 = svgToScreen(Math.max(marqueeStart.x,p.x), Math.max(marqueeStart.y,p.y));
  marqueeEl.setAttribute("x",s1.x); marqueeEl.setAttribute("y",s1.y);
  marqueeEl.setAttribute("width",s2.x-s1.x); marqueeEl.setAttribute("height",s2.y-s1.y);
  return {x1:Math.min(marqueeStart.x,p.x), y1:Math.min(marqueeStart.y,p.y), x2:Math.max(marqueeStart.x,p.x), y2:Math.max(marqueeStart.y,p.y)};
}
function finishMarquee(box){
  const hits = [];
  Array.from(drawLayer.children).forEach(el=>{
    const bb = getVisualBBox(el);
    if(bb.x>=box.x1 && bb.y>=box.y1 && bb.x+bb.width<=box.x2 && bb.y+bb.height<=box.y2){
      hits.push(el);
    }
  });
  setSelection(hits);
  marqueeEl && marqueeEl.remove();
  marqueeEl = null;
}

function openPersonalShapesDb(){
  if(personalShapeDbPromise) return personalShapeDbPromise;
  personalShapeDbPromise = new Promise((resolve,reject)=>{
    const req = indexedDB.open(PERSONAL_SHAPES_DB,2);
    req.onupgradeneeded = ()=>{
      const db=req.result;
      if(!db.objectStoreNames.contains(PERSONAL_SHAPES_STORE)){
        db.createObjectStore(PERSONAL_SHAPES_STORE,{keyPath:"id"});
      }
      if(!db.objectStoreNames.contains(LINKED_SHAPES_STORE)){
        db.createObjectStore(LINKED_SHAPES_STORE,{keyPath:"id"});
      }
    };
    req.onsuccess=()=>resolve(req.result);
    req.onerror=()=>reject(req.error || new Error("Could not open personal Shapes library"));
  });
  return personalShapeDbPromise;
}

async function ensureDirectoryPermission(handle,ask){
  if(!handle) return false;
  try{
    if((await handle.queryPermission({mode:"read"}))==="granted") return true;
  }catch(e){}
  if(ask){
    try{
      return (await handle.requestPermission({mode:"read"}))==="granted";
    }catch(e){}
  }
  return false;
}

async function scanLinkedSvgFolder(handle,prefix=""){
  const files=[];
  for await (const [name,entry] of handle.entries()){
    if(entry.kind==="file" && /\.svg$/i.test(name)){
      const f=await entry.getFile();
      files.push({name,path:prefix+name,svg:await f.text()});
    }else if(entry.kind==="directory"){
      const nested=await scanLinkedSvgFolder(entry,prefix+name+"/");
      files.push(...nested);
    }
  }
  return files;
}

function insertLibraryShapeText(txt,name){
  try{
    const doc=new DOMParser().parseFromString(txt,"image/svg+xml");
    const root=doc.documentElement;
    if(!root || root.tagName.toLowerCase()!=="svg") throw new Error("Invalid SVG");
    const vb=(root.getAttribute("viewBox")||"0 0 100 100").trim().split(/[\s,]+/).map(Number);
    const w=(vb[2]||100), h=(vb[3]||100);
    const target=150;
    const scale=Math.min(target/w,target/h);
    const cr=canvasWrap.getBoundingClientRect();
    const center=screenToSvg(cr.left+cr.width/2,cr.top+cr.height/2);

    const g=document.createElementNS(SVGNS,"g");
    g.id=newId();
    g.dataset.libraryShape=name;
    g.setAttribute("transform",`translate(${center.x-(w*scale)/2} ${center.y-(h*scale)/2}) scale(${scale}) translate(${-(vb[0]||0)} ${-(vb[1]||0)})`);

    const inheritedAttrs=[
      "fill","stroke","stroke-width","stroke-linecap","stroke-linejoin",
      "stroke-miterlimit","stroke-dasharray","stroke-dashoffset",
      "fill-rule","clip-rule","opacity","color","style"
    ];
    inheritedAttrs.forEach(attr=>{
      const v=root.getAttribute(attr);
      if(v!==null && v!=="") g.setAttribute(attr,v);
    });

    const rootStyleText=(root.getAttribute("style")||"")+" "+(root.getAttribute("stroke")||"")+" "+(root.getAttribute("fill")||"");
    if(/currentColor/i.test(rootStyleText) && !root.getAttribute("color")){
      g.setAttribute("color","#000000");
    }

    Array.from(root.childNodes).forEach(n=>{
      if(n.nodeType===1) g.appendChild(document.importNode(n,true));
    });

    drawLayer.appendChild(g);
    setSelection([g]);
    pushHistory();
    shapeStatus.textContent="Added "+name+" to Drawing canvas";
  }catch(err){
    shapeStatus.textContent="Shape failed: "+(err?.message||err);
  }
}

function texSettings(){
  return {size:+texSize.value,smoothing:+texSmooth.value,taper:+texTaper.value,roughness:+texRough.value,jitter:+texJitter.value};
}
function texDist(a,b){return Math.hypot(b.x-a.x,b.y-a.y)}
function texNorm(x,y){const m=Math.hypot(x,y)||1;return{x:x/m,y:y/m}}
function texRng(seed){let s=(seed|0)||1234567;return()=>{s=(s*1664525+1013904223)|0;return((s>>>0)/4294967296)}}
function texTaperFactor(i,n,taper){
  if(n<2) return 1;
  const u=i/(n-1);
  const edge=Math.min(u,1-u)*2;
  return (1-taper)+taper*Math.max(.04,Math.sin(Math.PI*Math.min(1,edge)));
}
function buildTextureOutline(points,settings){
  if(!points || points.length<2) return null;
  const sm=[points[0]];
  for(let i=1;i<points.length-1;i++){
    const a=points[i-1],b=points[i],c=points[i+1];
    sm.push({x:b.x+( (a.x+b.x+c.x)/3-b.x)*settings.smoothing,
             y:b.y+( (a.y+b.y+c.y)/3-b.y)*settings.smoothing,
             pressure:b.pressure||.5});
  }
  sm.push(points[points.length-1]);
  const rng=texRng(sm.length*911+Math.round(settings.roughness*1000)*17);
  const left=[],right=[];
  for(let i=0;i<sm.length;i++){
    const prv=sm[Math.max(0,i-1)],nxt=sm[Math.min(sm.length-1,i+1)];
    const t=texNorm(nxt.x-prv.x,nxt.y-prv.y), n={x:-t.y,y:t.x};
    const pressure=sm[i].pressure||.5;
    let width=settings.size*(.80+.35*pressure)*texTaperFactor(i,sm.length,settings.taper)*.5;
    const edgeNoise=settings.roughness*settings.size*(rng()-.5)*.9;
    const center=settings.jitter*settings.size*(rng()-.5)*.55;
    const cx=sm[i].x+n.x*center,cy=sm[i].y+n.y*center;
    const off=Math.max(.6,width+edgeNoise);
    left.push({x:cx+n.x*off,y:cy+n.y*off});
    right.push({x:cx-n.x*off,y:cy-n.y*off});
  }
  return left.concat(right.reverse());
}
function texturePolyToD(poly){
  if(!poly || !poly.length) return "";
  return "M "+poly.map(p=>p.x.toFixed(2)+" "+p.y.toFixed(2)).join(" L ")+" Z";
}

}

  const pts = elToPathPoints(el);
  if(!pts) return el;

  ["fill","stroke","stroke-width","opacity","transform"].forEach(attr=>{

    if(v!==null) path.setAttribute(attr, v); else path.removeAttribute(attr);
  });
  el.parentNode.insertBefore(path, el);
  el.remove();
  return path;
}

function pickElementAt(ev){
  const els = document.elementsFromPoint(ev.clientX, ev.clientY);
  for(const e of els){
    if(!e || e===drawLayer || e===svgRoot || e===pageRect || e===selOverlay) continue;

    let node = e;
    while(node && node.parentNode && node.parentNode !== drawLayer){
      if(node === svgRoot || node === document.body || node === document.documentElement){
        node = null;
        break;
      }
      node = node.parentNode;
    }

    if(node && node.parentNode === drawLayer){
      return node;
    }
  }
  return null;
}

function moveElement(el, dx, dy, startBB, startTransform){
  const m = /rotate\(([^ ]+) ([^ ]+) ([^ ]+)\)/.exec(startTransform);
  if(el.tagName==="rect"||el.tagName==="ellipse"||el.tagName==="text"){
    applyBBoxToElement(el, startBB.x+dx, startBB.y+dy, startBB.width, startBB.height);
  } else if(el.tagName==="line"){
    const x1=+el.dataset.ox1||+el.getAttribute("x1"), y1=+el.getAttribute("y1"), x2=+el.getAttribute("x2"), y2=+el.getAttribute("y2");
  }
  if(el.tagName==="line"){
    if(el._origLine===undefined) el._origLine = [+el.getAttribute("x1"),+el.getAttribute("y1"),+el.getAttribute("x2"),+el.getAttribute("y2")];
  }
  if(el.tagName==="path"){
    if(el._origD===undefined) el._origD = pathToPoints(el);
  }
  if(m){
    el.setAttribute("transform", `rotate(${m[1]} ${+m[2]+dx} ${+m[3]+dy})`);
  }
}

function getBooleanPaperScope(){
  try { return window.parent?.paper || window.parent?.paperScope || window.parent?.Wick?.paper || window.paper || null; }

function svgMatrixScale(m){
  if(!m) return 1;
  const sx=Math.hypot(Number(m.a)||0,Number(m.b)||0);
  const sy=Math.hypot(Number(m.c)||0,Number(m.d)||0);
  const vals=[sx,sy].filter(v=>isFinite(v)&&v>0);
  return vals.length ? vals.reduce((a,b)=>a+b,0)/vals.length : 1;
}

function bakeVisibleStrokeStyle(sourceRoot,cloneRoot){
  const srcNodes=[sourceRoot].concat(sourceRoot.querySelectorAll ? Array.from(sourceRoot.querySelectorAll("*")) : []);
  const dstNodes=[cloneRoot].concat(cloneRoot.querySelectorAll ? Array.from(cloneRoot.querySelectorAll("*")) : []);

  for(let i=0;i<Math.min(srcNodes.length,dstNodes.length);i++){
    const src=srcNodes[i], dst=dstNodes[i];
    const tag=(src.tagName||"").toLowerCase();
    if(!["path","rect","circle","ellipse","line","polyline","polygon"].includes(tag)) continue;

    let cs=null;
    try{ cs=getComputedStyle(src); }catch(e){}

    let stroke="";
    let fill="";
    let sw=0;

    try{
      stroke=(cs?.stroke && cs.stroke!=="none") ? cs.stroke : (src.getAttribute("stroke")||"");
      fill=(cs?.fill && cs.fill!=="none") ? cs.fill : (src.getAttribute("fill")||"");
      sw=parseFloat(cs?.strokeWidth || src.getAttribute("stroke-width") || "0") || 0;
    }catch(e){}

    if(stroke && stroke!=="none" && sw>0){
      let scale=1;
      try{ scale=svgMatrixScale(src.getCTM()); }catch(e){}
      const visualWidth=sw*scale;
      dst.setAttribute("stroke",stroke==="currentcolor"||stroke==="currentColor" ? "#000000" : stroke);
      dst.setAttribute("stroke-width",String(visualWidth));
      dst.setAttribute("vector-effect","non-scaling-stroke");
      const lc=cs?.strokeLinecap || src.getAttribute("stroke-linecap");
      const lj=cs?.strokeLinejoin || src.getAttribute("stroke-linejoin");
      if(lc) dst.setAttribute("stroke-linecap",lc);
      if(lj) dst.setAttribute("stroke-linejoin",lj);
    } else {
      dst.setAttribute("stroke","none");
      dst.setAttribute("stroke-width","0");
    }

    if(fill && fill!=="none"){
      dst.setAttribute("fill",fill==="currentcolor"||fill==="currentColor" ? "#000000" : fill);
    } else {
      dst.setAttribute("fill","none");
    }
  }
}

function serializeOneDrawingObject(el){
  const wrap = document.createElementNS(SVGNS,"svg");
  wrap.setAttribute("xmlns",SVGNS);
  wrap.setAttribute("width",pageW);
  wrap.setAttribute("height",pageH);
  wrap.setAttribute("viewBox",`0 0 ${pageW} ${pageH}`);

  const clone=el.cloneNode(true);
  bakeVisibleStrokeStyle(el,clone);
  wrap.appendChild(clone);

  return new XMLSerializer().serializeToString(wrap);
}

function booleanCollectPathItems(item){
  const out=[];
  (function walk(node){
    if(!node) return;
    if(node.className==="Path" || node.className==="CompoundPath"){
      out.push(node);
      return;
    }
    if(node.children?.length) Array.from(node.children).forEach(walk);
  })(item);
  return out;
}

function booleanUniteItems(items){
  if(!items?.length) return null;
  let acc=items[0].clone({insert:false});
  for(let i=1;i<items.length;i++){
    let merged=null;
    try{ merged=acc.unite(items[i],{insert:false}); }catch(e){ merged=null; }
    try{ acc.remove(); }catch(e){}
    if(!merged) return null;
    acc=merged;
  }
  return acc;
}

function paperEffectiveScale(item){
  try{
    const m=item.globalMatrix || item.matrix;
    if(!m) return 1;
    const sx=Math.hypot(Number(m.a)||1,Number(m.b)||0);
    const sy=Math.hypot(Number(m.c)||0,Number(m.d)||1);
    return Math.max(.0001,(sx+sy)/2);
  }catch(e){ return 1; }
}

function paperGlobalPoint(item,pt){
  try{
    if(typeof item.localToGlobal==="function") return item.localToGlobal(pt);
  }catch(e){}
  return pt.clone ? pt.clone() : pt;
}

function paperStrokeOutline(path,paper){
  const sw=Number(path.strokeWidth)||0;
  if(!path.strokeColor || sw<=0) return null;

  const radius=Math.max(.25,sw/2);
  const work=path.clone({insert:false});
  try{ work.flatten(Math.max(.12,Math.min(.25,radius*.02))); }catch(e){}

  const pts=(work.segments||[]).map(s=>paperGlobalPoint(work,s.point));
  const closed=!!work.closed;
  try{work.remove();}catch(e){}
  if(pts.length<2) return null;

  const parts=[];

  function addDisc(p){
    try{
      const c=new paper.Path.Circle({center:p,radius:radius,insert:false});
      c.fillColor="black";
      c.strokeColor=null;
      parts.push(c);
    }catch(e){}
  }

  function addSegment(a,b){
    const dx=b.x-a.x,dy=b.y-a.y;
    const len=Math.hypot(dx,dy);
    if(len<.001){ addDisc(a); return; }
    const nx=-dy/len*radius, ny=dx/len*radius;
    const poly=new paper.Path({
      segments:[
        [a.x+nx,a.y+ny],
        [b.x+nx,b.y+ny],
        [b.x-nx,b.y-ny],
        [a.x-nx,a.y-ny]
      ],
      closed:true,
      insert:false
    });
    poly.fillColor="black";
    poly.strokeColor=null;
    parts.push(poly);
  }

  for(let i=0;i<pts.length-1;i++){
    addSegment(pts[i],pts[i+1]);
    addDisc(pts[i]);
  }
  addDisc(pts[pts.length-1]);

  if(closed && pts.length>2) addSegment(pts[pts.length-1],pts[0]);

  const outline=booleanUniteItems(parts);
  parts.forEach(x=>{try{x.remove();}catch(e){}});
  if(outline){
    outline.fillColor="black";
    outline.strokeColor=null;
  }
  return outline;
}

function paperFilledGeometry(path){
  if(!path.fillColor) return null;
  try{
    const c=path.clone({insert:false});
    if(c.globalMatrix && typeof c.transform==="function"){
      try{ c.transform(c.globalMatrix); }catch(e){}
    }
    c.strokeColor=null;
    c.fillColor="black";
    return c;
  }catch(e){ return null; }
}

function booleanVisibleGeometry(paths,paper){
  const visible=[];
  for(const path of paths){
    const fill=paperFilledGeometry(path);
    if(fill) visible.push(fill);
    const stroke=paperStrokeOutline(path,paper);
    if(stroke) visible.push(stroke);
  }
  const item=booleanUniteItems(visible);
  visible.forEach(x=>{try{x.remove();}catch(e){}});
  return item;
}

function booleanImportObject(el,paper){
  const svg=serializeOneDrawingObject(el);
  const imported=paper.project.importSVG(svg,{insert:false,expandShapes:true});
  if(!imported) return {root:null,item:null};
  const paths=booleanCollectPathItems(imported);
  const item=booleanVisibleGeometry(paths,paper);
  return {root:imported,item};
}

function importBooleanResultToDrawing(svgText){
  const doc = new DOMParser().parseFromString(svgText,"image/svg+xml");
  let node = doc.documentElement;
  if(!node) throw new Error("Boolean result could not be converted back to SVG");

  function forceFilledGeometry(root){
    const all=[root].concat(root.querySelectorAll ? Array.from(root.querySelectorAll("*")) : []);
    all.forEach(n=>{
      const tag=(n.tagName||"").toLowerCase();
      if(["path","compoundpath","rect","circle","ellipse","polygon","polyline"].includes(tag)){
        n.setAttribute("fill","#000000");
        n.setAttribute("stroke","none");
        n.removeAttribute("stroke-width");
        n.removeAttribute("stroke-linecap");
        n.removeAttribute("stroke-linejoin");
        const style=n.getAttribute("style")||"";
        if(style){
          const cleaned=style
            .replace(/fill\s*:\s*[^;]+;?/gi,"")
            .replace(/stroke\s*:\s*[^;]+;?/gi,"")
            .replace(/stroke-width\s*:\s*[^;]+;?/gi,"");
          n.setAttribute("style","fill:#000000;stroke:none;"+cleaned);
        }
      }
    });
  }

  if(node.tagName.toLowerCase()==="svg"){
    forceFilledGeometry(node);
    const kids = Array.from(node.children).filter(n=>!["defs","title","desc"].includes(n.tagName.toLowerCase()));
    if(kids.length===1) node=kids[0];
    else{
      const g=document.createElementNS(SVGNS,"g");
      kids.forEach(k=>g.appendChild(document.importNode(k,true)));
      g.id=newId();
      g.setAttribute("fill","#000000");
      g.setAttribute("stroke","none");
      drawLayer.appendChild(g);
      return g;
    }
  } else {
    forceFilledGeometry(node);
  }

  const imported=document.importNode(node,true);
  imported.id=newId();
  imported.setAttribute("fill","#000000");
  imported.setAttribute("stroke","none");
  drawLayer.appendChild(imported);
  return imported;
}

function runBooleanOperation(op){
  if(selected.length!==2){
    flashStatus("Boolean: select exactly two objects");
    return;
  }

  const paper=getBooleanPaperScope();
  if(!paper?.project?.importSVG){
    flashStatus("Boolean: Paper.js is not available in this Wick build");
    return;
  }

  const aEl=selected[0], bEl=selected[1];
  let aRoot=null,bRoot=null,a=null,b=null,result=null;

  try{
    const ai=booleanImportObject(aEl,paper);
    const bi=booleanImportObject(bEl,paper);
    aRoot=ai.root; bRoot=bi.root; a=ai.item; b=bi.item;

    if(!a || !b) throw new Error("Use closed vector shapes with fill");

    if(op==="unite") result=a.unite(b,{insert:false});
    else if(op==="subtract") result=a.subtract(b,{insert:false});
    else if(op==="intersect") result=a.intersect(b,{insert:false});
    else if(op==="exclude") result=a.exclude(b,{insert:false});
    else throw new Error("Unknown Boolean operation");

    if(!result) throw new Error("No Boolean result");

    try{
      result.fillColor="black";
      result.strokeColor=null;
      if(result.children?.length){
        result.children.forEach(ch=>{try{ch.fillColor="black";ch.strokeColor=null;}catch(e){}});
      }
    }catch(e){}
    const resultSvg=result.exportSVG({asString:true});
    aEl.remove();
    bEl.remove();
    const newEl=importBooleanResultToDrawing(resultSvg);
    setSelection([newEl]);
    pushHistory();
    flashStatus("Boolean " + op + " complete");
  }catch(err){
    console.error("[Drawing] Boolean failed",err);
    flashStatus("Boolean failed: " + (err?.message || err));
  }finally{
    try{result?.remove();}catch(e){}
    try{a?.remove();}catch(e){}
    try{b?.remove();}catch(e){}
    try{aRoot?.remove();}catch(e){}
    try{bRoot?.remove();}catch(e){}
  }
}

function smoothPaperPaths(root,amount){
  const paths=booleanCollectPathItems(root);
  let changed=0;

  // The user-facing amount is deliberately scaled down.
  // Paper.js simplify() uses document units, so feeding 1.25 directly
  // was aggressive enough to visibly deform small LukeTools shapes.
  const tolerance=Math.max(.02,Math.min(1.0,Number(amount||1)*.08));

  paths.forEach(p=>{
    try{
      if(typeof p.simplify!=="function") return;

      const oldBounds=p.bounds?.clone ? p.bounds.clone() : null;
      const oldCenter=oldBounds?.center?.clone ? oldBounds.center.clone() : null;

      const ok=p.simplify(tolerance);
      if(ok===false) return;

      // Keep the overall footprint stable. This makes Smooth clean facets
      // rather than changing the designed proportions of the object.
      if(oldBounds && oldCenter && p.bounds){
        const nb=p.bounds;
        const sx=(nb.width>0) ? oldBounds.width/nb.width : 1;
        const sy=(nb.height>0) ? oldBounds.height/nb.height : 1;

        if(isFinite(sx)&&isFinite(sy)&&sx>0&&sy>0){
          try{ p.scale(sx,sy,p.bounds.center); }catch(e){}
        }
        try{ p.position=oldCenter; }catch(e){}
      }

      changed++;
    }catch(e){}
  });
  return changed;
}

function replaceSelectedWithPaperSvg(root){
  if(!selected.length) throw new Error("Nothing selected");
  const old=selected.slice();
  const svgText=root.exportSVG({asString:true});
  const doc=new DOMParser().parseFromString(svgText,"image/svg+xml");
  let node=doc.documentElement;
  if(!node) throw new Error("Could not export smoothed SVG");

  if(node.tagName.toLowerCase()==="svg"){
    const kids=Array.from(node.children).filter(n=>!["defs","title","desc"].includes(n.tagName.toLowerCase()));
    if(kids.length===1){
      node=kids[0];
    }else{
      const g=document.createElementNS(SVGNS,"g");
      kids.forEach(k=>g.appendChild(document.importNode(k,true)));
      node=g;
    }
  }

  const imported=document.importNode(node,true);
  imported.id=newId();
  old.forEach(el=>el.remove());
  drawLayer.appendChild(imported);
  setSelection([imported]);
  pushHistory();
  return imported;
}

function normalizeToCubic(pts){
  let anchor = null;
  const out = [];
  for(let i=0;i<pts.length;i++){
    const seg = pts[i];
    if(seg.type==="M"){ anchor = {x:seg.x,y:seg.y}; out.push(Object.assign({},seg)); continue; }
    if(seg.type==="Z"){ out.push(Object.assign({},seg)); continue; }
    if(seg.type==="Q"){
      const P0=anchor, C={x:seg.c1x,y:seg.c1y}, P1={x:seg.x,y:seg.y};
      const c1={x:P0.x+(C.x-P0.x)*2/3, y:P0.y+(C.y-P0.y)*2/3};
      const c2={x:P1.x+(C.x-P1.x)*2/3, y:P1.y+(C.y-P1.y)*2/3};
      out.push({type:"C", c1x:c1.x,c1y:c1.y,c2x:c2.x,c2y:c2.y,x:P1.x,y:P1.y});
      anchor = P1;
      continue;
    }
    out.push(Object.assign({},seg));
    anchor = {x:seg.x, y:seg.y};
  }
  return out;
}
function ptsToPaperSegments(pts){
  const norm = normalizeToCubic(pts);
  const anchors = [];
  let prevAnchor = null;
  for(let i=0;i<norm.length;i++){
    const seg = norm[i];
    if(seg.type==="M"){ const a={x:seg.x,y:seg.y,hIn:null,hOut:null}; anchors.push(a); prevAnchor=a; continue; }
    if(seg.type==="Z"){ continue; }
    if(seg.type==="L"){ const a={x:seg.x,y:seg.y,hIn:null,hOut:null}; anchors.push(a); prevAnchor=a; continue; }
    if(seg.type==="C"){
      if(prevAnchor) prevAnchor.hOut = {x:seg.c1x,y:seg.c1y};
      const a = {x:seg.x, y:seg.y, hIn:{x:seg.c2x,y:seg.c2y}, hOut:null};
      anchors.push(a);
      prevAnchor = a;
    }
  }
  const closed = pts.some(p=>p.type==="Z");
  const segments = anchors.map(a=>{
    const hi = a.hIn ? [a.hIn.x-a.x, a.hIn.y-a.y] : [0,0];
    const ho = a.hOut ? [a.hOut.x-a.x, a.hOut.y-a.y] : [0,0];
    if(hi[0]===0 && hi[1]===0 && ho[0]===0 && ho[1]===0) return [a.x,a.y];
    return [a.x,a.y, hi[0],hi[1], ho[0],ho[1]];
  });
  return {segments, closed};
}
function hexToRgbArr(hex){
  if(!hex || hex==="none") return null;
  const h = hex.replace("#","");
  if(h.length<6) return null;
  return [parseInt(h.substring(0,2),16)/255, parseInt(h.substring(2,4),16)/255, parseInt(h.substring(4,6),16)/255, 1];
}
function elToPathPoints(el){
  if(el.tagName==="path") return pathToPoints(el);
  if(el.tagName==="rect"){
    const x=+el.getAttribute("x"), y=+el.getAttribute("y"), w=+el.getAttribute("width"), h=+el.getAttribute("height");
    return [{type:"M",x,y},{type:"L",x:x+w,y},{type:"L",x:x+w,y:y+h},{type:"L",x,y:y+h},{type:"Z"}];
  }
  if(el.tagName==="ellipse"){
    const cx=+el.getAttribute("cx"), cy=+el.getAttribute("cy"), rx=+el.getAttribute("rx"), ry=+el.getAttribute("ry");
    const k=0.5522847498;
    return [
      {type:"M", x:cx+rx, y:cy},
      {type:"C", c1x:cx+rx, c1y:cy+ry*k, c2x:cx+rx*k, c2y:cy+ry, x:cx, y:cy+ry},
      {type:"C", c1x:cx-rx*k, c1y:cy+ry, c2x:cx-rx, c2y:cy+ry*k, x:cx-rx, y:cy},
      {type:"C", c1x:cx-rx, c1y:cy-ry*k, c2x:cx-rx*k, c2y:cy-ry, x:cx, y:cy-ry},
      {type:"C", c1x:cx+rx*k, c1y:cy-ry, c2x:cx+rx, c2y:cy-ry*k, x:cx+rx, y:cy},
      {type:"Z"}
    ];
  }
  if(el.tagName==="line"){
    return [{type:"M", x:+el.getAttribute("x1"), y:+el.getAttribute("y1")}, {type:"L", x:+el.getAttribute("x2"), y:+el.getAttribute("y2")}];
  }
  return null;
}

function getActiveWickFrame(editor){
  const project = editor?.project;
  if(!project) return null;

  let frameNum = 1;
  try {
    if(typeof project.currentFrameNumber === "number") frameNum = Math.max(1, project.currentFrameNumber);
    else {
      const tl0 = project.activeTimeline || project.timeline;
      if(tl0?.playheadPosition != null) frameNum = Math.max(1, Number(tl0.playheadPosition)||1);
    }
  } catch(e){}

  try {
    const focus = project.focus || project.activeObject;
    if(focus?.classname === "Frame" || focus?._classname === "Frame") return focus;
  } catch(e){}

  try {
    const tl = project.activeTimeline || project.timeline || project.root?.timeline;
    if(!tl) return null;
    if(tl.activeFrame) return tl.activeFrame;
    if(typeof tl.getActiveFrame === "function"){
      const f = tl.getActiveFrame();
      if(f) return f;
    }

    let layer = tl.activeLayer;
    if(typeof layer === "number" && tl.layers) layer = tl.layers[layer] || tl.layers[0];
    if(!layer && tl.layers?.length) layer = tl.layers[0];
    if(!layer) return null;

    if(typeof layer.getFrame === "function"){
      const f = layer.getFrame(frameNum);
      if(f) return f;
    }
    if(typeof layer.getFrameAtPlayheadPosition === "function"){
      const f = layer.getFrameAtPlayheadPosition(frameNum);
      if(f) return f;
    }
    if(layer.activeFrame) return layer.activeFrame;
    if(layer.frames?.length){
      for(const f of layer.frames){
        if(f && f.start <= frameNum && (f.end >= frameNum || f.start === frameNum)) return f;
      }
      return layer.frames[0];
    }
  } catch(e){}

  return null;
}

function addWickPathToFrame(frame, obj){
  if(typeof frame.addPath === "function"){ frame.addPath(obj); return; }
  if(typeof frame.addObject === "function"){ frame.addObject(obj); return; }
  if(frame.clip?.addPath){ frame.clip.addPath(obj); return; }
  if(Array.isArray(frame.paths)){ frame.paths.push(obj); return; }
  if(Array.isArray(frame.content)){ frame.content.push(obj); return; }
  throw new Error("No Wick frame insertion API");
}

}

function prepareDrawingCloneForDemBones(sourceEl){
  const clone=sourceEl.cloneNode(true);

  // Remove editor-only XML blocks that have no visual meaning in DemBones and
  // commonly contain application-specific namespace attributes.
  try{
    const remove=[];
    if(clone.querySelectorAll){
      clone.querySelectorAll("sodipodi\\:namedview,namedview,metadata").forEach(node=>remove.push(node));
    }
    remove.forEach(node=>{
      try{ node.remove(); }
      catch(e){ try{ node.parentNode&&node.parentNode.removeChild(node); }catch(e2){} }
    });
  }catch(e){}

  return clone;
}

function addDemBonesSvgNamespaces(wrapper){
  const XMLNS="http://www.w3.org/2000/xmlns/";
  const namespaces={
    "xmlns:inkscape":"http://www.inkscape.org/namespaces/inkscape",
    "xmlns:sodipodi":"http://sodipodi.sourceforge.net/DTD/sodipodi-0.dtd",
    "xmlns:xlink":"http://www.w3.org/1999/xlink",
    "xmlns:dc":"http://purl.org/dc/elements/1.1/",
    "xmlns:cc":"http://creativecommons.org/ns#",
    "xmlns:rdf":"http://www.w3.org/1999/02/22-rdf-syntax-ns#"
  };
  Object.entries(namespaces).forEach(([name,value])=>{
    try{ wrapper.setAttributeNS(XMLNS,name,value); }
    catch(e){ try{ wrapper.setAttribute(name,value); }catch(e2){} }
  });
}

function selectedSvgDocumentForDemBones(){

  let minX=Infinity,minY=Infinity,maxX=-Infinity,maxY=-Infinity;

    minY=Math.min(minY,bb.y);
    maxX=Math.max(maxX,bb.x+bb.width);
    maxY=Math.max(maxY,bb.y+bb.height);
  });

  if(!isFinite(minX)||!isFinite(minY)||!isFinite(maxX)||!isFinite(maxY)){
    minX=0;
    minY=0;
    maxX=Math.max(1,pageW);
    maxY=Math.max(1,pageH);

  minX-=pad;
  minY-=pad;
  maxX+=pad;
  maxY+=pad;

  const h=Math.max(1,maxY-minY);

  wrapper.setAttribute("height",h);

function createDrawingSelectionWickPathsForDemBones(){
  if(!selected.length) throw new Error("Select one or more Drawing objects first");

  const editor=getWickEditor();
  const paper=getWickPaper();
  if(!editor?.project) throw new Error("Wick editor not found");
  if(!paper?.project?.importSVG) throw new Error("Paper.js import is not available in this Wick build");

  const frame=getActiveWickFrame(editor);
  if(!frame) throw new Error("No active Wick frame found");

  const svgText=selectedSvgDocument();
  if(!svgText) throw new Error("Nothing selected");

  const project=editor.project;
  const created=[];
  let imported=null;
  let paperError=null;

  try{
    try{
      imported=paper.project.importSVG(svgText,{insert:false,expandShapes:true});
    }catch(ePaperImport){
      paperError=ePaperImport;
      imported=null;
    }

    if(imported){
      const wickW=Number(project.width||project.stageWidth||pageW)||pageW;
      const wickH=Number(project.height||project.stageHeight||pageH)||pageH;
      const sx=wickW/Math.max(1,pageW);
      const sy=wickH/Math.max(1,pageH);

      if((Math.abs(sx-1)>0.0001||Math.abs(sy-1)>0.0001)&&imported.scale){
        imported.scale(sx,sy,new paper.Point(0,0));
      }

      const paperPaths=collectPaperPaths(imported);
      for(const pItem of paperPaths){
        try{
          const wickPath=createWickPathFromPaperItem(pItem);
          addWickPathToFrame(frame,wickPath);
          created.push(wickPath);
        }catch(ePath){
          console.warn("[Drawing] DemBones Wick path insert skipped an item",ePath);
        }
      }
    }

    if(!created.length){
      const wickW=Number(project.width||project.stageWidth||pageW)||pageW;
      const wickH=Number(project.height||project.stageHeight||pageH)||pageH;
      const sx=wickW/Math.max(1,pageW);
      const sy=wickH/Math.max(1,pageH);

      selected.forEach(root=>{
        drawingGeometryElements(root).forEach(el=>{
          let pp=null;
          try{
            pp=drawingElementToPaperPath(el,paper,sx,sy);
            if(!pp) return;
            const wickPath=createWickPathFromPaperItem(pp);
            addWickPathToFrame(frame,wickPath);
            created.push(wickPath);
          }catch(eDirect){
            console.warn("[Drawing] DemBones direct Wick conversion skipped an element",eDirect);
          }finally{
            try{ if(pp) pp.remove(); }catch(eRemove){}
          }
        });
      });
    }

    if(!created.length){
      if(paperError) console.warn("[Drawing] Paper SVG import failed before direct DemBones conversion",paperError);
      throw new Error("Selected Drawing objects could not be converted to Wick paths");
    }

    try{
      if(project.markAsModified) project.markAsModified();
      project.unsavedChanges=true;
      if(editor.canvas?.render) editor.canvas.render();
      if(editor.projectDidChange) editor.projectDidChange();
      window.parent?.dispatchEvent(new Event("resize"));
    }catch(eRefresh){}

    return created;
  }finally{
    try{ if(imported) imported.remove(); }catch(eRemoveImported){}
  }
}

function collectPaperPaths(item){
  const out = [];
  function walk(node){
    if(!node) return;
    if(node.className === "Path" || node.className === "CompoundPath"){
      out.push(node);
      return;
    }
    if(node.children?.length){
      Array.from(node.children).forEach(walk);
    }
  }
  walk(item);
  return out;
}

function createWickPathFromPaperItem(paperItem){
  const Wick = window.parent?.Wick || window.Wick;
  if(!Wick?.Path) throw new Error("Wick.Path is not available");

  const json = paperItem.exportJSON({asString:false});
  let wp = null;

  try { wp = new Wick.Path({json}); }
  catch(e1){
    try { wp = new Wick.Path(json, paperItem); }
    catch(e2){ wp = new Wick.Path(json); }
  }

  try { if(wp.updateJSON) wp.updateJSON(); } catch(e){}
  return wp;
}

function drawingColorForPaper(value){
  value=String(value||"").trim();
  if(!value || value==="none" || value==="transparent") return null;
  if(value==="currentcolor" || value==="currentColor") return "#000000";
  return value;
}

function drawingGeometryElements(root){
  const out=[];
  function walk(node){
    if(!node || node.nodeType!==1) return;
    const tag=(node.tagName||"").toLowerCase();
    if(["path","rect","circle","ellipse","line","polyline","polygon"].includes(tag)){
      out.push(node);
      return;
    }
    Array.from(node.children||[]).forEach(walk);
  }
  walk(root);
  return out;
}

function drawingElementClosed(el){
  const tag=(el.tagName||"").toLowerCase();
  if(["rect","circle","ellipse","polygon"].includes(tag)) return true;
  if(tag==="path") return /z\s*$/i.test(String(el.getAttribute("d")||"").trim()) || /z/i.test(String(el.getAttribute("d")||""));
  return false;
}

function drawingElementToPaperPath(el,paper,sx,sy){
  if(!el || typeof el.getTotalLength!=="function" || typeof el.getPointAtLength!=="function") return null;
  let total=0;
  try { total=Number(el.getTotalLength())||0; } catch(e){ total=0; }
  if(!(total>0)) return null;

  const ctm=el.getCTM ? el.getCTM() : null;
  function mapPoint(pt){
    let x=Number(pt.x)||0, y=Number(pt.y)||0;
    if(ctm){
      const nx=ctm.a*x + ctm.c*y + ctm.e;
      const ny=ctm.b*x + ctm.d*y + ctm.f;
      x=nx; y=ny;
    }
    return new paper.Point(x*sx,y*sy);
  }

  const samples=Math.max(8,Math.min(320,Math.ceil(total/3)));
  const pp=new paper.Path();
  for(let i=0;i<=samples;i++){
    const d=total*(i/samples);
    let p=null;
    try { p=el.getPointAtLength(d); } catch(e){}
    if(!p) continue;
    pp.add(mapPoint(p));
  }
  if(pp.segments && pp.segments.length<2){
    try { pp.remove(); } catch(e){}
    return null;
  }

  pp.closed=drawingElementClosed(el);

  let cs=null;
  try { cs=getComputedStyle(el); } catch(e){}
  const fill=drawingColorForPaper((cs&&cs.fill)||el.getAttribute("fill"));
  const stroke=drawingColorForPaper((cs&&cs.stroke)||el.getAttribute("stroke"));
  let sw=parseFloat((cs&&cs.strokeWidth)||el.getAttribute("stroke-width")||"0")||0;
  const matrixScale=ctm ? ((Math.hypot(ctm.a,ctm.b)+Math.hypot(ctm.c,ctm.d))/2) : 1;
  const stageScale=(Math.abs(sx)+Math.abs(sy))/2;

  try { pp.fillColor=fill; } catch(e){ pp.fillColor=null; }
  try { pp.strokeColor=stroke; } catch(e){ pp.strokeColor=null; }
  pp.strokeWidth=Math.max(0,sw*matrixScale*stageScale);

  try {
    const op=parseFloat((cs&&cs.opacity)||el.getAttribute("opacity")||"1");
    if(isFinite(op)) pp.opacity=Math.max(0,Math.min(1,op));
  } catch(e){}

  return pp;
}

function addDrawingSelectionDirectToWick(selectedEls,paper,frame,project){
  const wickW=Number(project.width||project.stageWidth||pageW)||pageW;
  const wickH=Number(project.height||project.stageHeight||pageH)||pageH;
  const sx=wickW/Math.max(1,pageW);
  const sy=wickH/Math.max(1,pageH);
  let count=0;

  selectedEls.forEach(root=>{
    drawingGeometryElements(root).forEach(el=>{
      let pp=null;
      try {
        pp=drawingElementToPaperPath(el,paper,sx,sy);
        if(!pp) return;
        const wickPath=createWickPathFromPaperItem(pp);
        addWickPathToFrame(frame,wickPath);
        count++;
      } catch(err){
        console.warn("[Drawing] Direct Wick conversion skipped an element",err);
      } finally {
        try { pp&&pp.remove(); } catch(e){}
      }
    });
  });
  return count;
}

    return;
  }

  btn.disabled = true;
  const oldText = btn.textContent;
  btn.textContent = "Adding…";

  let imported = null;
  try {
    const project = editor.project;

    // Preferred route: convert the Drawing selection into native Wick.Path objects.
    try {
      imported = paper.project.importSVG(svgText,{insert:false,expandShapes:true});
    } catch(ePaperImport) {
      paperError = ePaperImport;
      imported = null;
    }

    if(imported){

      const wickH = Number(project.height || project.stageHeight || pageH) || pageH;

      const paperPaths = collectPaperPaths(imported);
      let count = 0;
      for(const pItem of paperPaths){

          const wickPath = createWickPathFromPaperItem(pItem);
          addWickPathToFrame(frame,wickPath);

          if(project.markAsModified) project.markAsModified();

          if(editor.canvas?.render) editor.canvas.render();

    const directCount=addDrawingSelectionDirectToWick(selected,paper,frame,project);

        if(project.markAsModified) project.markAsModified();
        project.unsavedChanges = true;

      flashStatus("Added " + directCount + " path" + (directCount===1 ? "" : "s") + " to Wick stage");
      return;
    }

    // Last fallback for Wick builds that expose the project SVG importer.
    if(typeof project.importSVG === "function"){
      project.importSVG(svgText);
      try {
        if(project.markAsModified) project.markAsModified();
        project.unsavedChanges = true;
        if(editor.canvas?.render) editor.canvas.render();
        if(editor.projectDidChange) editor.projectDidChange();
        window.parent?.dispatchEvent(new Event("resize"));
      } catch(e){}
      flashStatus("Added selection to Wick stage");
      return;
    }

    if(paperError) console.warn("[Drawing] Paper SVG import failed before direct conversion",paperError);
    throw new Error("Selected Drawing object could not be converted to Wick paths");
  } catch(err){
    console.error("[Drawing] Add to Wick failed",err);
    flashStatus("Add to Wick failed: " + (err?.message || err));
  } finally {
    try { imported?.remove(); } catch(e){}
    btn.disabled = false;
    btn.textContent = oldText;
  }
}

function deleteDrawingFrame(index){

    return;
  }
  captureCurrentDrawingFrame();
  index=Math.max(0,Math.min(drawingFrames.length-1,Number(index)||0));
  drawingFrames.splice(index,1);
  if(drawingFrameIndex>=drawingFrames.length) drawingFrameIndex=drawingFrames.length-1;
  else if(index<drawingFrameIndex) drawingFrameIndex--;
  drawingFrames.forEach((frame,i)=>{ if(!String(frame.label||"").trim()) frame.label="Frame "+(i+1); });
  drawLayer.innerHTML=String(drawingFrames[drawingFrameIndex]?.drawing||"");
  resetDrawingHistoryForFrame();
  renderDrawingFrames();
  renderDrawingOnionSkin();
  scheduleDrawingSessionSave();
}

function drawGizmoIdentifier(value,fallback){
  let s=String(value||"").trim().replace(/[^A-Za-z0-9_$]+/g,"_").replace(/^_+|_+$/g,"");
  if(!s) s=fallback||"DrawGizmo";
  if(!/^[A-Za-z_$]/.test(s)) s="Draw_"+s;
  return s;
}

function drawingRemoveInternalFrames(layer){

  try{frames=Array.from(layer?.frames||layer?._children||[]);}catch(e){}
  for(let i=frames.length-1;i>=0;i--){
    const frame=frames[i];
    try{
      if(typeof layer.removeFrame==="function"){ layer.removeFrame(frame); continue; }
    }catch(e){}
    try{
      if(typeof frame.remove==="function"){ frame.remove(); continue; }
    }catch(e){}
    try{
      const list=layer.frames||layer._children;
      const at=list.indexOf(frame);
      if(at>=0) list.splice(at,1);
    }catch(e){}
  }
}

function drawingAddClipToFrame(frame,clip){

    const list=frame._children||frame.children;
    if(Array.isArray(list)){list.push(clip);try{clip._parent=frame;}catch(e){}return true;}
  }catch(e){}
  return false;
}

function drawingBuildNativeFrameSvg(snapshotHtml){ // FIX GPT_20-09-2026_NativeGizmoNamespaceSafe
  const drawing=String(snapshotHtml||"");
  if(!drawing.trim()) return "";

  // Build a real SVG DOM wrapper with all namespaces that Drawing/imported
  // artwork may legitimately contain. The previous bare string wrapper caused
  // Paper.js importSVG to reject frames containing prefixed SVG attributes.
  const wrapper=document.createElementNS(SVGNS,"svg");
  wrapper.setAttribute("xmlns",SVGNS);
  addDemBonesSvgNamespaces(wrapper);
  wrapper.setAttribute("width",String(pageW));
  wrapper.setAttribute("height",String(pageH));
  wrapper.setAttribute("viewBox","0 0 "+pageW+" "+pageH);

  const holder=document.createElementNS(SVGNS,"g");
  holder.innerHTML=drawing;

  // Remove only non-visual editor metadata. Do not alter paths, transforms,
  // fills, strokes or Bezier data.
  try{
    const remove=[];
    holder.querySelectorAll("metadata,namedview,sodipodi\\:namedview").forEach(node=>remove.push(node));
    remove.forEach(node=>{
      try{node.remove();}
      catch(e){try{node.parentNode&&node.parentNode.removeChild(node);}catch(e2){}}
    });
  }catch(e){}

  while(holder.firstChild) wrapper.appendChild(holder.firstChild);

  return new XMLSerializer().serializeToString(wrapper);
}

function drawingAddSnapshotToWickFrame(snapshotHtml,wickFrame,paper,project){
  const drawing=String(snapshotHtml||"");
  if(!drawing.trim()) return 0;
  if(!paper?.project?.importSVG) throw new Error("Paper.js SVG import is unavailable");

  const svgText=drawingBuildNativeFrameSvg(drawing);
  if(!svgText) return 0;

  let imported=null;

  try{
    try{
      imported=paper.project.importSVG(svgText,{insert:false,expandShapes:true});
    }catch(importError){
      console.error("[Drawing Native Gizmo] Paper import failed",importError,svgText.slice(0,800));
      throw new Error("Paper.js could not import this Drawing frame");
    }

    if(!imported) throw new Error("Paper.js could not import this Drawing frame");

    const wickW=Number(project.width||project.stageWidth||pageW)||pageW;
    const wickH=Number(project.height||project.stageHeight||pageH)||pageH;
    const sx=wickW/Math.max(1,pageW);
    const sy=wickH/Math.max(1,pageH);

    if((Math.abs(sx-1)>0.0001||Math.abs(sy-1)>0.0001)&&typeof imported.scale==="function"){
      imported.scale(sx,sy,new paper.Point(0,0));
    }

    const paperPaths=collectPaperPaths(imported);
    if(!paperPaths.length) throw new Error("Drawing frame contains no native vector paths");

    let count=0;
    paperPaths.forEach(pItem=>{
      const wickPath=createWickPathFromPaperItem(pItem);
      addWickPathToFrame(wickFrame,wickPath);
      count++;
    });

    if(!count) throw new Error("Drawing frame could not be converted to native Wick paths");
    return count;
  }finally{
    try{imported?.remove();}catch(e){}
  }
}

}


async function makeDrawGizmo(){ // FIX GPT_20-09-2026_CanonicalFrameGizmo
  const button=document.getElementById("btnTimelineMakeGizmo");
  captureCurrentDrawingFrame();

  const nonEmpty=drawingFrames.filter(frame=>String(frame.drawing||"").trim());
  if(!nonEmpty.length){
    flashStatus("Draw at least one frame before making a Gizmo");
    return;
  }

  const editor=getWickEditor();
  const paper=getWickPaper();
  const project=editor?.project;
  const W=drawingWickNamespace();

  if(!project||!W?.Clip||!W?.Frame||!paper?.project?.importSVG){
    flashStatus("Native Wick Clip/Frame/Paper APIs are unavailable");
    return;
  }

  const stageFrame=getActiveWickFrame(editor);
  if(!stageFrame){
    flashStatus("No active Wick frame found");
    return;
  }

  const requestedName=drawGizmoIdentifier(
    document.getElementById("drawGizmoName")?.value,
    "DrawGizmo"
  );

  const oldText=button?.textContent||"Make Gizmo";
  if(button){
    button.disabled=true;
    button.textContent=drawingCalledFromDemBones() ? "Building native Gizmo…" : "Building…";
  }

  let clip=null;
  let addedToStage=false;

  try{
    try{clip=new W.Clip({project});}
    catch(e1){try{clip=new W.Clip();}catch(e2){}}
    if(!clip) throw new Error("Could not create native Wick Clip");

    try{clip.identifier=requestedName;}catch(e){}
    try{clip.name=requestedName;}catch(e){}
    try{clip._identifier=requestedName;}catch(e){}

    const layer=drawingCreateInternalLayer(clip,project);
    if(!layer) throw new Error("Could not create the Gizmo internal layer");
    drawingRemoveInternalFrames(layer);

    let totalPaths=0;

    drawingFrames.forEach((drawingFrame,index)=>{
      const wickFrame=new W.Frame({
        start:index+1,
        length:1,
        project
      });

      const label=drawGizmoIdentifier(
        drawingFrame.label,
        "Frame_"+(index+1)
      );

      try{wickFrame.identifier=label;}catch(e){}
      try{wickFrame.name=label;}catch(e){}

      if(typeof layer.addFrame!=="function"){
        throw new Error("Wick layer.addFrame is unavailable");
      }

      layer.addFrame(wickFrame);

      // Canonical rule: Draw Gizmo frames are native Wick paths created from
      // exact Paper SVG geometry. Never fall back to sampled line geometry.
      totalPaths+=drawingAddSnapshotToWickFrame(
        drawingFrame.drawing,
        wickFrame,
        paper,
        project
      );
    });

    if(totalPaths<=0){
      throw new Error("Drawing frames contained no native Wick paths");
    }

    // This is the exact marker used by LukeTools FrameGizmo.
    const familyId=String(clip.uuid||"").replace(/[^A-Za-z0-9_-]/g,"") ||
      ("gizmo_"+Date.now()+"_"+Math.floor(Math.random()*1000000));
    const marker=
      "/*LUKE_FRAME_GIZMO:1*/\n"+
      "/*LUKE_FRAME_GIZMO_ID:"+familyId+"*/";

    try{
      if(typeof clip.addScript==="function"){
        clip.addScript("default",marker);
      }else{
        if(!Array.isArray(clip.scripts)) clip.scripts=[];
        clip.scripts.push({name:"default",src:marker});
      }
    }catch(e){
      if(!Array.isArray(clip.scripts)) clip.scripts=[];
      clip.scripts.push({name:"default",src:marker});
    }

    // Same default playback state as an ordinary LukeTools Gizmo.
    try{clip.isSynced=false;}catch(e){}
    try{clip.animationType="single";}catch(e){}
    try{clip.singleFrameNumber=1;}catch(e){}
    try{if(clip.timeline) clip.timeline.playheadPosition=1;}catch(e){}

    if(!drawingAddClipToFrame(stageFrame,clip)){
      throw new Error("Could not place the native Gizmo Clip on the Wick frame");
    }
    addedToStage=true;

    drawingMarkProjectDirty(editor,project,"Create LukeTools Gizmo");

    // Let Wick build the Clip view before FrameGizmo/DemBones reads it.
    try{if(project.view?.render) project.view.render();}catch(e){}
    try{if(editor.canvas?.render) editor.canvas.render();}catch(e){}
    await new Promise(resolve=>{
      const raf=window.parent?.requestAnimationFrame||window.requestAnimationFrame;
      if(typeof raf==="function") raf(()=>raf(resolve));
      else setTimeout(resolve,40);
    });

    try{
      const bridge=window.LukeToolsBridge||window.parent?.LukeToolsBridge;
      if(bridge?.selectObjectByInfo){
        bridge.selectObjectByInfo({
          uuid:String(clip.uuid||""),
          identifier:String(clip.identifier||requestedName),
          name:String(clip.name||requestedName)
        });
      }
    }catch(e){}

    if(drawingCalledFromDemBones()){
      const receiver=window.parent && window.parent.__LukeToolsDemBonesImportNativeGizmo;
      if(typeof receiver!=="function"){
        throw new Error("DemBones canonical Gizmo importer is not available");
      }

      let result=receiver({
        uuid:String(clip.uuid||""),
        removeSource:true
      });

      if(result&&typeof result.then==="function") result=await result;

      if(!result||result.ok===false){
        throw new Error(
          result&&result.error ?
            result.error :
            "DemBones did not accept the native Gizmo"
        );
      }

      try{
        if(window.parent.__LukeToolsDrawingLaunchContext){
          window.parent.__LukeToolsDrawingLaunchContext=null;
        }
      }catch(e){}

      try{
        const pd=window.parent?.document;
        const shell=pd?.getElementById("luketools_dembones_drawing_panel");
        if(shell?.parentNode) shell.parentNode.removeChild(shell);
      }catch(e){}

      flashStatus("Native Gizmo added to DemBones");
      return;
    }

    flashStatus(
      "Created LukeTools Gizmo “"+requestedName+"” with "+
      drawingFrames.length+" frame"+(drawingFrames.length===1?"":"s")
    );
  }catch(err){
    console.error("[Drawing] Make native LukeTools Gizmo failed",err);

    // Keep a successfully-created stage Clip only for independent Drawing.
    // When called from DemBones a failed handoff must not leave residue.
    if(drawingCalledFromDemBones()&&clip&&addedToStage){
      try{
        if(typeof clip.remove==="function") clip.remove();
        else if(clip.parentFrame?.removeClip) clip.parentFrame.removeClip(clip);
        else if(clip.parentFrame?.removeObject) clip.parentFrame.removeObject(clip);
      }catch(eRemove){}
      drawingMarkProjectDirty(editor,project,"Remove failed Draw Gizmo");
    }

    flashStatus("Gizmo failed: "+(err?.message||err));
  }finally{
    if(button){
      button.disabled=false;
      button.textContent=oldText;
    }
  }
}
